            tr: tr, name: v[0], amb: v[1],
            s: num(v[2]), c: num(v[3]), op: num(v[4]), ar: num(v[5]), inv: num(v[6]), ap: num(v[7]), sic: v[8]
          };
        }).filter(function (r) { return String(r.name || '').trim(); });
      }

      /* Registro de las sociedades incorporadas POR EL PUNTO DE INGESTA DE
         COMPARABLES LOCALES (base de la Superintendencia de Sociedades u otra
         fuente nacional). Sólo estas tienen carácter obligatorio.
         Una compañía de la base de Capital IQ constituida en Colombia NO entra
         aquí: se filtra con los mismos criterios que las demás. */
      window.PT_LOCALES = window.PT_LOCALES || [];
      window.PT_LOCALES_KEYS = window.PT_LOCALES_KEYS || {};

      function registrarLocal(nombre) {
        if (!nombre) return;
        var k = nameKey(nombre);
        if (!k || window.PT_LOCALES_KEYS[k]) return;
        window.PT_LOCALES_KEYS[k] = true;
        window.PT_LOCALES.push(nombre);
      }
      window.ptRegistrarComparableLocal = registrarLocal;

      function esLocal(r) {
        return !!window.PT_LOCALES_KEYS[nameKey(r.name)];
      }
      window.ptEsComparableLocal = esLocal;

      function cuartil(orden, q) {
        var n = orden.length; if (!n) return null;
        var pos = q * (n - 1), lo = Math.floor(pos), hi = Math.min(lo + 1, n - 1);
        return orden[lo] + (orden[hi] - orden[lo]) * (pos - lo);
      }

      window.ptFiltrarAtipicos = function (modo) {
        var K = window.PT_CRITERIOS;
        modo = modo || K.atipicos;
        var kind = ($('pli') && $('pli').value) || 'MO';
        var filas = filasTabla();
        if (filas.length < 4) { moToast('Se requieren al menos 4 comparables para depurar valores atípicos.', 'amber'); return null; }

        var conPLI = filas.map(function (r) {
          var pli = pliOf({ s: r.s, c: r.c, op: r.op }, kind);
          return { r: r, pli: pli, local: esLocal(r) };
        }).filter(function (x) { return x.pli !== null && isFinite(x.pli); });

        var valores = conPLI.map(function (x) { return x.pli; }).sort(function (a, b) { return a - b; });
        var q1 = cuartil(valores, 0.25), q3 = cuartil(valores, 0.75), riq = q3 - q1;
        var limInf = null, limSup = null, criterio = '';

        if (modo === 'iqr') {
          limInf = q1 - K.atipicosFactor * riq; limSup = q3 + K.atipicosFactor * riq;
          criterio = 'Regla del rango intercuartílico: se excluyen las observaciones situadas más de ' + K.atipicosFactor +
            ' veces el recorrido intercuartílico por debajo del primer cuartil o por encima del tercero (límites ' +
            pctf(limInf) + ' y ' + pctf(limSup) + ').';
        } else if (modo === 'percentil') {
          limInf = cuartil(valores, 0.05); limSup = cuartil(valores, 0.95);
          criterio = 'Recorte por percentiles: se excluyen las observaciones por debajo del percentil 5 (' + pctf(limInf) +
            ') y por encima del percentil 95 (' + pctf(limSup) + ').';
        } else if (modo === 'umbral') {
          limInf = -K.atipicosUmbral / 100; limSup = K.atipicosUmbral / 100;
          criterio = 'Umbral absoluto: se excluyen las compañías cuyo indicador de rentabilidad excede ±' + K.atipicosUmbral +
            ' %, por corresponder a resultados que no reflejan una operación rutinaria comparable.';
        } else {
          criterio = 'No se aplicó depuración de valores atípicos: el rango se calcula sobre todas las comparables aceptadas.';
        }

        if (typeof MOTOR === 'undefined') return null;
        MOTOR.activo = true;
        MOTOR.sel = MOTOR.sel || {}; MOTOR.motivo = MOTOR.motivo || {};

        var excluidas = [], conservadas = [], forzadas = [];
        conPLI.forEach(function (x) {
          var k = nameKey(x.r.name);
          var fuera = (limInf !== null) && (x.pli < limInf || x.pli > limSup);
          if (fuera && x.local) {
            /* las comparables locales son de inclusión obligatoria: se conservan
               aunque su margen sea atípico, dejando constancia expresa */
            MOTOR.sel[k] = true;
            MOTOR.motivo[k] = 'Comparable local de inclusión obligatoria; su indicador (' + pctf(x.pli) + ') queda fuera de los límites de depuración pero se conserva por compartir el mercado geográfico de la parte examinada.';
            forzadas.push({ name: x.r.name, pli: x.pli });
            conservadas.push({ name: x.r.name, pli: x.pli, local: true });
            return;
          }
          if (fuera) {
            MOTOR.sel[k] = false;
            MOTOR.motivo[k] = 'Excluida por indicador atípico (' + pctf(x.pli) + '). ' + criterio;
            excluidas.push({ name: x.r.name, pli: x.pli, id: (window.COMP_ID || {})[x.r.name] || '' });
          } else {
            MOTOR.sel[k] = true;
            conservadas.push({ name: x.r.name, pli: x.pli, local: x.local });
          }
        });

        window.PT_ATIPICOS = {
          modo: modo, criterio: criterio, limInf: limInf, limSup: limSup,
          q1: q1, q3: q3, riq: riq, kind: kind,
          excluidas: excluidas, conservadas: conservadas, forzadas: forzadas,
          n: conPLI.length, fecha: new Date().toISOString()
        };

        try { calc(); } catch (e) { }
        ptPintarAtipicos();
        moToast(excluidas.length + ' comparable(s) excluidas por margen atípico; quedan ' + conservadas.length + '.', excluidas.length ? 'amber' : 'green');
        return window.PT_ATIPICOS;
      };

      window.ptMarcarLocalesObligatorias = function () {

          n++;

        if (n) { try { calc(); } catch (e) { } }
        return n;

      ['integrarSuper', 'integrarSuperAuto'].forEach(function (fn) {

          } catch (e) { }
          return r;
        };

      var SYN_COMP = {
        id: ['excel company id', 'company id', 'capital iq id', 'ciq id', 'identificador'],

      function colDe(headers, claves) {
        for (var i = 0; i < headers.length; i++) {
          var h = String(headers[i] || '').toLowerCase().trim();
          if (claves.some(function (k) { return h.indexOf(k) >= 0; })) return i;
        }
        return -1;
      }

      window.ptIngestarEEFFComparables = async function (file) {
        if (!file) return null;
        var rows;
        try {
          if (/\.(xlsx?|xlsm)$/i.test(file.name)) {
            await loadXLSX();
            var wb = XLSX.read(new Uint8Array(await file.arrayBuffer()), { type: 'array' });
            var hoja = wb.SheetNames[0];
            wb.SheetNames.forEach(function (n) { if (/eeff|financ|screening|datos/i.test(n)) hoja = n; });
            rows = XLSX.utils.sheet_to_json(wb.Sheets[hoja], { header: 1, defval: '' });
          } else {
            var txt = await file.text();
            rows = txt.split(/\r?\n/).map(function (l) { return l.split(/\t|;|,/); });
          }
        } catch (e) { moToast('No se pudo leer el archivo: ' + e.message, 'red'); return null; }

        var hi = (typeof findHeaderRow === 'function') ? findHeaderRow(rows) : 0;
        var headers = (rows[hi] || []).map(String);
        var col = {}; Object.keys(SYN_COMP).forEach(function (k) { col[k] = colDe(headers, SYN_COMP[k]); });
        if (col.s < 0 && col.op < 0) { moToast('El archivo no contiene columnas de ingresos ni de utilidad operacional reconocibles.', 'red'); return null; }

        /* índice de la tabla actual por identificador y por nombre */
        var filas = filasTabla();
        var porId = {}, porNombre = {};
        filas.forEach(function (r) {
          porNombre[nameKey(r.name)] = r;
          var id = (window.COMP_ID || {})[r.name];
          if (id) porId[String(id).toUpperCase()] = r;
        });

        var act = 0, sinCruce = [], conBal = 0;
        for (var i = hi + 1; i < rows.length; i++) {
          var f = rows[i] || [];
          var id = col.id >= 0 ? String(f[col.id] || '').trim().toUpperCase() : '';
          var nom = col.name >= 0 ? String(f[col.name] || '').trim() : '';
          if (!id && !nom) continue;
          var destino = (id && porId[id]) || (nom && porNombre[nameKey(nom)]) || null;
          if (!destino) { if (nom || id) sinCruce.push(nom || id); continue; }

          var inputs = destino.tr.querySelectorAll('input');
          /* orden de los campos numéricos en la fila: s, c, op, ar, inv, ap */
          var mapa = [['s', 1], ['c', 2], ['op', 3], ['ar', 4], ['inv', 5], ['ap', 6]];
          var tocada = false;
          mapa.forEach(function (m) {
            var ci = col[m[0]];
            if (ci < 0) return;
            var v = String(f[ci] === undefined ? '' : f[ci]).trim();
            if (v === '') return;
            var n = parseFloat(v.replace(/[^0-9.\-]/g, ''));
            if (!isFinite(n)) return;
            inputs[m[1]].value = n;
            tocada = true;
          });
          if (tocada) act++;
          if (col.ar >= 0 && col.ap >= 0 && String(f[col.ar]).trim() !== '' && String(f[col.ap]).trim() !== '') conBal++;

          /* se refleja también en las estructuras del motor */
          try {
            var meta = (window.PT_CRIBA && PT_CRIBA.filas || []).filter(function (x) { return nameKey(x.name) === nameKey(destino.name); })[0];
            if (meta) {
              if (col.s >= 0) meta.s = num(inputs[1].value);
              if (col.c >= 0) meta.c = num(inputs[2].value);
              if (col.op >= 0) meta.op = num(inputs[3].value);
              if (col.ar >= 0) meta.ar = num(inputs[4].value);
              if (col.inv >= 0) meta.inv = num(inputs[5].value);
              if (col.ap >= 0) meta.ap = num(inputs[6].value);
            }
          } catch (e) { }
        }

        window.PT_EEFF_COMP = {
          cargado: new Date().toISOString(), archivo: file.name,
          actualizadas: act, total: filas.length, sinCruce: sinCruce.slice(0, 12), conBalance: conBal
        };

        try { calc(); } catch (e) { }
        try { if (window.PT_CRITERIOS.atipicos !== 'ninguno') ptFiltrarAtipicos(); } catch (e) { }
        ptPintarEeffComp();
        moToast('EEFF de comparables incorporados: ' + act + ' de ' + filas.length +
          (conBal ? ' · ' + conBal + ' con cuentas de balance (el ajuste de capital de trabajo ya es practicable)' : ''), 'green');
        return window.PT_EEFF_COMP;
      };

            body: JSON.stringify({ model: 'claude-haiku-4-5', max_tokens: 8000, messages: [{ role: 'user', content: prompt }] })
          });
          if (!res.ok) throw new Error('HTTP ' + res.status);
          data = await res.json();
        }
        if (data.error) throw new Error(data.error.message || 'error de la API');
        var txt = (data.content || []).filter(function (b) { return b.type === 'text'; }).map(function (b) { return b.text; }).join('\n');
        return extraerJSONDeRespuestaIA(txt);
      }

      /* Control: ninguna cifra del texto puede ser ajena al contexto */
      function verificarCifras(texto, contexto) {
        var nums = String(texto).match(/\b\d[\d.,]{2,}\b/g) || [];
        var ctx = String(contexto).replace(/[.,\s]/g, '');
        var ajenas = [];
        nums.forEach(function (n) {
          var limpio = n.replace(/[.,]/g, '');
          if (limpio.length < 3) return;
          if (ctx.indexOf(limpio) < 0 && ajenas.indexOf(n) < 0) ajenas.push(n);
        });
        return ajenas;
      }
      window.ptVerificarCifras = verificarCifras;

      /* ───────────────────────────────────────────────────────────────────────
         6. PROCESO

      window.ptRedactarInforme = async function () {
        var host = document.getElementById('ptRedaccion');
        var pintar = function (h) { if (host) host.innerHTML = h; };

        var P = ptPendientes();
        if (!P.total) { pintar('<div class="alert-green" style="font-size:11.5px">No hay apartados pendientes: el informe está completo.</div>'); return null; }

        pintar('<div style="font-size:11.5px;color:#1F618D">Redactando ' + P.total + ' apartado(s) con la información disponible…</div>');
        var contexto = ptContextoCompleto();
        var pendientes = P.identidad.concat(P.economico, P.sector);

        var resultado = { campos: {}, via: '', pendientes: P, fecha: new Date().toISOString(), alertas: [] };

        /* vía IA */
        try {
          var j = await redaccionIA(pendientes, contexto);
          resultado.via = 'asistida';
          Object.keys(j.campos || {}).forEach(function (id) {
            var c = j.campos[id];
            if (!c || !c.texto) return;
            var ajenas = verificarCifras(c.texto, contexto);
            resultado.campos[id] = {
              texto: c.texto, fuente: c.fuente || '', normas: c.normas || [],
              suficiente: c.suficiente !== false, cifrasAjenas: ajenas
            };
            if (ajenas.length) resultado.alertas.push('El apartado «' + id + '» contiene cifras que no constan en la información cargada: ' + ajenas.join(', '));
          });
        } catch (e) {
          /* vía determinista */
          resultado.via = 'local';
          resultado.error = e.message;
          pendientes.forEach(function (c) {
            resultado.campos[c[0]] = {
              texto: redaccionLocal(c[0], c[1]),
              fuente: (window.PT_ANTERIOR && PT_ANTERIOR.fuente) || 'información cargada en el estudio',
              normas: [], suficiente: true, cifrasAjenas: []
            };
          });
        }

        window.PT_REDACCION = resultado;
        ptPintarRedaccion();
        return resultado;
      };

      window.ptAplicarRedaccion = function () {
        var R = window.PT_REDACCION; if (!R) return;
        var n = 0, faltantes = 0;
        Object.keys(R.campos).forEach(function (id) {
          var e = $(id); if (!e) return;
          if (String(e.value || '').trim()) return;          // nunca se sobrescribe lo diligenciado
          var c = R.campos[id];
          /* texto sin sustento real (la propia redacción avisa que falta la información):
             no se escribe en el campo — se deja pendiente, en vez de imprimir el aviso
             como si fuera el dato. Coincide con el criterio ya usado en ptPintarRedaccion. */
          if (c.suficiente === false || /no contiene informaci|pendiente de completar|debe solicitarse/i.test(c.texto)) { faltantes++; return; }
          e.value = c.texto; n++;

        R.aplicado = new Date().toISOString(); R.nAplicados = n;
        try { calc(); } catch (e) { }
        moToast(n + ' apartado(s) diligenciados automáticamente' + (faltantes ? '; ' + faltantes + ' quedaron pendientes por falta de información (no se inventaron)' : '') + '. Revíselos antes de radicar.', 'green');
        ptPintarRedaccion();

    /* ═══════════════════════════════════════════════════════════════════════════
       PARCHE PT — Bloque 11:  EXTENSIÓN Y FORMA DEL INFORME ANTERIOR
    
       El informe del año gravable debe guardar continuidad con el del período
       precedente: mismas secciones, en el mismo orden y con una extensión
       comparable. Un informe notablemente más breve que el del año anterior es
       difícil de sostener ante un requerimiento, porque sugiere que apartados
       que antes se desarrollaron ahora se despacharon en dos líneas.
    
       Este bloque lee el informe anterior, levanta su esqueleto —secciones y
       extensión de cada una—, lo compara con el informe en curso y fija la
       extensión objetivo que la redacción proactiva debe alcanzar.
       ═════════════════════════════════════════════════════════════════════════ */
    (function () {
      'use strict';

      window.PT_ESTRUCTURA = null;

      var ESQUELETO = [
        { id: 'intro', nom: 'Introducción', chars: 2000, rx: /introducci[oó]n/i },
        { id: 'resumen', nom: 'Resumen ejecutivo', chars: 6600, rx: /resumen ejecutivo/i },
        { id: 'general', nom: 'Información general y vinculación económica', chars: 6400, rx: /informaci[oó]n general|vinculaci[oó]n econ[oó]mica|descripci[oó]n de la compa/i },

        { id: 'metodo', nom: 'Selección del método, comparables y depuración', chars: 15500, rx: /selecci[oó]n del (mejor )?m[eé]todo|compa[ñn][ií]as comparables|proceso filtro|razones de rechazo|metodolog[ií]a de selecci[oó]n|conjunto de comparables/i },
        { id: 'rango', nom: 'Resultados: indicador, ajustes y rango intercuartil', chars: 4400, rx: /resultados y conclusiones|indicador de rentabilidad|rango intercuartil|ajustes a las compa/i },

      /* ───────────────────────────────────────────────────────────────────────
         1. LEVANTAR EL ESQUELETO DEL INFORME ANTERIOR
         ─────────────────────────────────────────────────────────────────────── */
      var DENSIDAD = 1830;   // caracteres por página medidos en informes completos

      window.ptEstructuraAnterior = function () {
        var A = window.PT_ANTERIOR;
        if (!A || !A.texto) { moToast('Cargue primero el informe del año anterior.', 'amber'); return null; }

        var toc = [], fuente = 'esqueleto de referencia';
        /* la tabla de contenido trae líneas «Título …… página» */
        var lineas = A.texto.slice(0, 9000).split(/\n/);
        lineas.forEach(function (l) {
          /* Los puntos de la guía suelen venir separados por espacios cuando el
             PDF se extrae («Título ..... ..... ..... 12»); por eso el separador se
             admite como una mezcla de puntos y espacios, exigiendo un mínimo de
             puntos para no confundirlo con una frase corriente. */
          var m = l.match(/^\s*(.{6,90}?)([.\s]{5,})(\d{1,3})\s*$/);
          if (m && (m[2].match(/\./g) || []).length >= 3) {
            toc.push({ titulo: m[1].replace(/\s+/g, ' ').trim(), pag: +m[3] });
          }
        });

        var secciones = [];
        if (toc.length >= 6) {
          fuente = 'tabla de contenido del informe anterior';
          /* Sólo títulos de PRIMER NIVEL: numeración romana o los encabezados
             estándar. Los subtítulos se descartan; de lo contrario los tramos de
             página se solapan y la extensión se cuenta varias veces. */
          var esNivel1 = function (t) {
            return /^[IVXL]+\s*\./.test(t) || /^(introducci[oó]n|resumen ejecutivo|anexos?)\b/i.test(t);
          };
          var n1 = toc.filter(function (t) { return esNivel1(t.titulo); })
            .sort(function (a, b) { return a.pag - b.pag; });

          /* se elimina el duplicado por página y se conserva el primero */
          var vistas = {}, limpio = [];
          n1.forEach(function (t) { if (!vistas[t.pag]) { vistas[t.pag] = 1; limpio.push(t); } });

          if (limpio.length >= 5) {
            limpio.forEach(function (t, i) {
              var sig = limpio[i + 1];
              var fin = sig ? sig.pag : t.pag + 2;
              secciones.push({
                titulo: t.titulo, pagIni: t.pag, pagFin: fin, paginas: Math.max(1, fin - t.pag),
                esAnexo: /anexos?/i.test(t.titulo)
              });
            });
          }
        }
        /* extensión: exacta si se conservaron las páginas; proporcional si no */
        var totalChars = A.texto.replace(/\s+/g, ' ').length;
        var paginas = A.paginas || null;
        secciones.forEach(function (s) {
          if (paginas && paginas.length) {
            var off = paginas.length - (secciones[secciones.length - 1].pagFin || 1); // desfase portada/índice
            var ini = Math.max(0, s.pagIni + (off > 0 ? off : 8) - 1);
            var fin = Math.min(paginas.length, s.pagFin + (off > 0 ? off : 8) - 1);
            var txt = paginas.slice(ini, fin).join(' ');
            s.chars = txt.replace(/\s+/g, ' ').length;
            s.medicion = 'exacta';
          } else {
            /* Sin el texto página por página se estima con la densidad habitual
               de este tipo de informes. No depende de cuánta documentación
               adicional se haya concatenado al texto del informe anterior. */
            s.chars = s.esAnexo ? 0 : Math.round((s.paginas || 0) * DENSIDAD);
            s.medicion = 'estimada a razón de ' + DENSIDAD + ' caracteres por página';
          }
        });

        /* si no se pudo levantar, se usa el esqueleto de referencia */
        if (!secciones.length) {
          secciones = ESQUELETO.filter(function (e) { return e.chars; }).map(function (e) {
            return { titulo: e.nom, chars: e.chars, paginas: Math.round(e.chars / 1800), medicion: 'referencia' };
          });
        }

        var mapa = {};
        ESQUELETO.forEach(function (e) {
          var hit = secciones.filter(function (s) { return e.rx.test(s.titulo); });
          var chars = hit.reduce(function (a, s) { return a + (s.chars || 0); }, 0);
          mapa[e.id] = { nom: e.nom, objetivo: chars || e.chars, titulosPrevios: hit.map(function (s) { return s.titulo; }) };
        });

        var cuerpo = secciones.filter(function (s) { return !s.esAnexo; });
        var anexos = secciones.filter(function (s) { return s.esAnexo; });
        window.PT_ESTRUCTURA = {
          fuente: fuente, secciones: secciones, mapa: mapa,
          totalPrevio: cuerpo.reduce(function (a, s) { return a + (s.chars || 0); }, 0),
          paginasPrevias: cuerpo.reduce(function (a, s) { return a + (s.paginas || 0); }, 0),
          paginasAnexos: anexos.reduce(function (a, s) { return a + (s.paginas || 0); }, 0),
          fecha: new Date().toISOString()
        };
        ptPintarEstructura();
        return window.PT_ESTRUCTURA;
      };

      /* ───────────────────────────────────────────────────────────────────────
         2. COMPARAR EL INFORME EN CURSO CONTRA ESA EXTENSIÓN
         ─────────────────────────────────────────────────────────────────────── */
      window.ptCompararExtension = function () {
        var E = window.PT_ESTRUCTURA || ptEstructuraAnterior();
        if (!E) return null;
        var doc = $('doc'); if (!doc) return null;

        /* se mide cada sección del informe actual: se recorren los hijos directos
           de #doc en su orden real y se acumula el texto bajo el encabezado que
           esté vigente. Antes solo se buscaba «h2.rep», pero casi todo el cuerpo
           del informe se titula con «h3.rep» (o con un h2 sin esa clase, en el
           caso del esqueleto mínimo) — por eso el conteo daba siempre 0. Un
           «h3.rep» cuenta como su propia sección y, además, su texto se suma a la
           «h2.rep» que lo contiene, para no perder la agregación por tema. */
        var hijos = [].slice.call(doc.children);
        var actuales = [];
        var curH2 = null, curH3 = null;
        hijos.forEach(function (el) {
          var esRep = el.classList && el.classList.contains('rep');
          if (esRep && el.tagName === 'H2') {
            curH2 = { titulo: el.textContent.replace(/^\d+\.\s*/, '').trim(), chars: 0 };
            actuales.push(curH2);
            curH3 = null;
          } else if (esRep && el.tagName === 'H3') {
            curH3 = { titulo: el.textContent.replace(/^\d+\.\s*/, '').trim(), chars: 0 };
            actuales.push(curH3);
          } else {
            var len = (el.textContent || '').replace(/\s+/g, ' ').length;
            if (curH3) curH3.chars += len;
            if (curH2) curH2.chars += len;
          }
        });

        var filas = ESQUELETO.filter(function (e) { return e.chars; }).map(function (e) {
          var obj = (E.mapa[e.id] && E.mapa[e.id].objetivo) || e.chars;
          var hit = actuales.filter(function (a) { return e.rx.test(a.titulo); });
          var act = hit.reduce(function (a, x) { return a + x.chars; }, 0);
          var razon = obj ? act / obj : 0;
          return {
            id: e.id, nom: e.nom, objetivo: obj, actual: act, razon: razon,
            estado: act === 0 ? 'ausente' : razon < 0.5 ? 'muy_corta' : razon < 0.8 ? 'corta' : razon > 2 ? 'excede' : 'conforme',
            titulos: hit.map(function (x) { return x.titulo; })
          };
        });

        var totObj = filas.reduce(function (a, f) { return a + f.objetivo; }, 0);
        var totAct = filas.reduce(function (a, f) { return a + f.actual; }, 0);

        window.PT_EXTENSION = {
          filas: filas, totalObjetivo: totObj, totalActual: totAct,
          razonGlobal: totObj ? totAct / totObj : 0,
          paginasEstimadas: Math.round(totAct / 1800), paginasPrevias: E.paginasPrevias || Math.round(totObj / 1800),
          fecha: new Date().toISOString()
        };
        ptPintarEstructura();
        return window.PT_EXTENSION;
      };

      /* ───────────────────────────────────────────────────────────────────────
         3. LA REDACCIÓN PROACTIVA APUNTA A ESA EXTENSIÓN
         ─────────────────────────────────────────────────────────────────────── */
      var CAMPO_A_SECCION = {
        ec_vision: 'general', ec_prod: 'general', ec_ingr: 'general', ec_reparto: 'general',
        ec_geo: 'general', ec_valor: 'funcional', ec_fin: 'general', ec_intang: 'funcional',
        ec_mercado: 'general', ec_estrategia: 'general',
        sc_def: 'economia', sc_tam: 'economia', sc_comp: 'economia', sc_prec: 'economia',
        sc_dem: 'economia', sc_reg: 'economia', sc_ries: 'economia', sc_persp: 'economia', sc_fuentes: 'economia',
        e_mundo: 'economia', e_col: 'economia', e_sec: 'economia',
        objeto: 'general', oper: 'operacion', plijust: 'metodo'
      };
      var CAMPOS_POR_SECCION = {};
      Object.keys(CAMPO_A_SECCION).forEach(function (k) {
        var s = CAMPO_A_SECCION[k];
        (CAMPOS_POR_SECCION[s] = CAMPOS_POR_SECCION[s] || []).push(k);
      });

      /* ───────────────────────────────────────────────────────────────────────
         2-bis. CONTENIDO DEL NEGOCIO Y DEL SECTOR EN EL INFORME
         Las tarjetas «Informe económico extendido» (pasos D y E del expediente)
         recogen la información del negocio del grupo (campos ec_) y del sector
         (campos sc_ y e_), pero las funciones que arman ese HTML (ecoExtHTML,
         estudioSectorHTML) nunca se insertaban en #doc: el contribuyente
         diligenciaba los campos y ese texto no llegaba al informe. Por eso el

         línea ~2426) mostraban siempre 0 en las secciones más grandes del

         ─────────────────────────────────────────────────────────────────────── */
      window.ptContenidoNegocioSectorHTML = function () {
        var gral = ECO_SECS.filter(function (s) { return CAMPO_A_SECCION[s[0]] !== 'funcional'; });
        var func = ECO_SECS.filter(function (s) { return CAMPO_A_SECCION[s[0]] === 'funcional'; });
        var out = '<h2 class="rep">Información general y vinculación económica</h2>' + ecoExtHTML(gral);

          '<h3 class="rep">Panorama de la economía colombiana</h3><p>' + (ec ? escH(ec).replace(/\n/g, '</p><p>') : '[Pendiente: panorama de la economía colombiana del año gravable, con fuente y fecha de consulta.]') + '</p>' +
          estudioSectorHTML();
        return out;
      };

      function ptTextoPlanoAHtmlSeccion(texto) {
        if (!texto || !String(texto).trim()) return null;
        var lineas = String(texto).split(/\n+/).map(function (l) { return l.trim(); }).filter(Boolean);
        if (!lineas.length) return null;
        var html = lineas.map(function (linea) {
          if (/^[I|V|X|L]+\.\s+|^\d+\.\s+/i.test(linea) || (linea.length < 80 && /^[A-ZÁÉÍÓÚÑ\s\.\,\:\-\(\)]{5,}$/.test(linea))) {
            return '<h3 class="rep">' + escH(linea) + '</h3>';
          }
          return '<p class="norm">' + escH(linea) + '</p>';
        }).join('');
        return html;
      }

      function ptPosicionParaSeccion(doc, secId) {
        var idx = ESQUELETO.findIndex(function (e) { return e.id === secId; });
        if (idx < 0) return null;

        if (secId === 'intro') {
          var h1 = doc.querySelector('h1');
          if (h1) return { padre: doc, ref: h1.nextSibling };
        }

        var headings = [].slice.call(doc.querySelectorAll('h2.rep, h3.rep, h2'));
        for (var i = idx + 1; i < ESQUELETO.length; i++) {
          var sig = ESQUELETO[i];
          var match = headings.find(function (h) { return sig.rx.test(h.textContent); });
          if (match) return { padre: match.parentNode, ref: match };
        }
        return null;
      }

      function ptAplicarSeccionesInyectadas() {
        var doc = $('doc'); if (!doc) return;
        var inyectadas = window.PT_SECCIONES_INYECTADAS || {};
        var keys = Object.keys(inyectadas);
        if (!keys.length) return;

        keys.forEach(function (secId) {
          var item = ESQUELETO.find(function (e) { return e.id === secId; });
          if (!item) return;
          var htmlContent = inyectadas[secId];
          if (!htmlContent) return;

          /* se retira primero cualquier inyección previa de esta misma
             sección — incluido su propio encabezado de emergencia, si lo
             tenía — para no buscar luego un «targetHeading» que esté a
             punto de eliminarse (eso dejaba el nodo sin padre y abortaba
             a mitad el forEach de todas las secciones pendientes). */
          var prevInjs = [].slice.call(doc.querySelectorAll('.inj-' + secId));
          prevInjs.forEach(function (el) { if (el.parentNode) el.parentNode.removeChild(el); });

          var headings = [].slice.call(doc.querySelectorAll('h2.rep, h3.rep, h2'));
          var targetHeading = headings.find(function (h) { return item.rx.test(h.textContent); });

          var tempDiv = document.createElement('div');
          tempDiv.innerHTML = htmlContent;
          var nodes = [].slice.call(tempDiv.childNodes);
          nodes.forEach(function (node) {
            if (node.nodeType === 1) node.classList.add('inj-' + secId);
          });

          if (targetHeading) {
            var next = targetHeading.nextElementSibling;
            if (next && next.tagName === 'P' && /\[Pendiente:/i.test(next.textContent)) {
              next.parentNode.removeChild(next);
            }
            var ref = targetHeading.nextSibling;
            nodes.forEach(function (node) {
              if (ref) targetHeading.parentNode.insertBefore(node, ref);
              else targetHeading.parentNode.appendChild(node);
            });
          } else {
            var h2 = document.createElement('h2');
            h2.className = 'rep inj-' + secId;
            h2.textContent = item.nom;

            var porInsertar = [h2].concat(nodes);
            var pos = ptPosicionParaSeccion(doc, secId);
            porInsertar.forEach(function (node) {
              if (pos) pos.padre.insertBefore(node, pos.ref);
              else doc.appendChild(node);
            });
          }
        });
      }

          throw new Error(msg);

        return data;
      };

      window.ptTextoYFuentes = function (data) {

          var A = window.PT_ANTERIOR; if (!A || !A.texto) return '';
          var mapa = { e_mundo: /econom[ií]a mundial/i, e_col: /econom[ií]a colombiana/i, e_sec: /an[aá]lisis del sector/i };
          var esSubSector = id.indexOf('sc_') === 0;
          var etiqueta = mapa[id] || (esSubSector ? mapa.e_sec : null);
          if (!etiqueta) return '';

      /* Extrae el JSON de la respuesta aunque el modelo haya añadido narración,
         preguntas o pasos intermedios antes o después del objeto final: nunca se
         debe volcar ese texto de proceso al informe. */
      /* Los modelos redactan prosa larga dentro del campo "texto" del JSON y con
         frecuencia dejan saltos de línea crudos dentro de la cadena — inválidos
         para JSON.parse, aunque el contenido en sí sea correcto. Antes de
         descartarlo se reintenta escapando los caracteres de control que
         aparezcan DENTRO de cadenas (nunca fuera, para no tocar la sintaxis). */
      function _escaparControlEnCadenas(s) {
        var out = '', dentro = false, escapado = false;
        for (var i = 0; i < s.length; i++) {
          var c = s[i];
          if (dentro) {
            if (escapado) { out += c; escapado = false; continue; }
            if (c === '\\') { out += c; escapado = true; continue; }
            if (c === '"') { dentro = false; out += c; continue; }
            if (c === '\n') { out += '\\n'; continue; }
            if (c === '\r') { out += '\\r'; continue; }
            if (c === '\t') { out += '\\t'; continue; }
            out += c; continue;
          }
          if (c === '"') { dentro = true; }
          out += c;
        }
        return out;
      }
      function _intentarJSON(s) {
        try { return JSON.parse(s); } catch (e) { }
        try { return JSON.parse(_escaparControlEnCadenas(s)); } catch (e) { return null; }
      }
      function extraerJSON(texto) {
        var t = String(texto || '').replace(/```json|```/g, '').trim();
        var r = _intentarJSON(t);
        if (r) return r;
        /* Recorre cada posible «{» inicial de atrás hacia adelante (con
           decremento simple: la versión anterior recalculaba lastIndexOf desde
           i-1, que con i=0 se queda clavada en 0 para siempre — bucle infinito
           que congela la pestaña en cuanto JSON.parse fallaba una sola vez). El
           conteo de llaves ignora las que aparezcan dentro de cadenas, para no
           cortar el objeto en un «{» o «}» que en realidad es solo texto. */
        for (var i = t.length - 1; i >= 0; i--) {
          if (t[i] !== '{') continue;
          var profundidad = 0, fin = -1, dentroStr = false, esc = false;
          for (var k = i; k < t.length; k++) {
            var c = t[k];
            if (dentroStr) {
              if (esc) { esc = false; }
              else if (c === '\\') { esc = true; }
              else if (c === '"') { dentroStr = false; }
              continue;
            }
            if (c === '"') { dentroStr = true; continue; }
            if (c === '{') profundidad++;
            else if (c === '}') { profundidad--; if (profundidad === 0) { fin = k; break; } }
          }
          if (fin > i) {
            r = _intentarJSON(t.slice(i, fin + 1));
            if (r) return r;
          }
        }
        return null;
      }

       Una compañía suele tener varias operaciones con vinculados. La ley las
       trata distinto según su cuantía:
    
         · Las que superan el tope por tipo de operación —45.000 UVT con
           vinculados del exterior, 10.000 UVT con jurisdicciones no cooperantes—
           deben DOCUMENTARSE: análisis funcional, método, comparables y rango

         · Las que no lo superan no se documentan, pero SÍ se declaran: el

       El sistema construía un único registro con los datos del formulario. Este

    (function () {

      /* ───────────────────────────────────────────────────────────────────────
         1. AGRUPACIÓN PARA LA DECLARACIÓN
         El Formato 1125 exige un registro por vinculado y tipo de operación.

      window.ptRegistrosDeclaracion = function () {
        var O = window.PT36 && PT36.ops;
        if (!O || !O.filas || !O.filas.length) return null;

        var uvt = O.uvt || num($('uvt') && $('uvt').value) || 0;
        var reg = {};
        O.filas.forEach(function (f) {
          if (!f.vinculado) return;
          var clave = (f.nit || f.vinculado) + '||' + (f.tipo || f.seccion || 'SIN TIPO') + '||' + f.regimen;
          if (!reg[clave]) {
            reg[clave] = {
              vinculado: f.vinculado, nit: f.nit, pais: f.pais, ciudad: f.ciudad,
              tipo: f.tipo, origenTipo: f.origenTipo, cod: f.cod, formato: f.formato, concepto: f.concepto,
              seccion: f.seccion, regimen: f.regimen, topeUVT: f.topeUVT,
              esVinculado: f.esVinculado, monto: 0, deducible: 0, n: 0, filas: []
            };
          }
          var r = reg[clave];
          r.monto += f.monto || 0;
          if (f.deducible) r.deducible += f.deducible;
          r.n++;
          r.filas.push(f.fila);
          if (f.cod && !r.cod) r.cod = f.cod;
        });

        var lista = Object.keys(reg).map(function (k) {
          var r = reg[k];
          r.topeCOP = (r.topeUVT || 45000) * uvt;
          r.documentable = r.monto >= r.topeCOP;
          r.tratamiento = r.documentable ? 'Documentada' : 'Declarada sin documentar';
          r.naturaleza = /ingreso/i.test(r.seccion || '') ? 'Ingreso' : /egreso/i.test(r.seccion || '') ? 'Egreso' : 'Otra';
          return r;
        }).sort(function (a, b) { return b.monto - a.monto; });

        var conVinc = lista.filter(function (r) { return r.esVinculado !== false; });
        var terceros = lista.filter(function (r) { return r.esVinculado === false; });

        window.PT_DECLARACION = {
          uvt: uvt, registros: conVinc, terceros: terceros,
          documentables: conVinc.filter(function (r) { return r.documentable; }),
          informativas: conVinc.filter(function (r) { return !r.documentable; }),
          totalDeclarado: conVinc.reduce(function (a, r) { return a + r.monto; }, 0),
          totalDocumentado: conVinc.filter(function (r) { return r.documentable; }).reduce(function (a, r) { return a + r.monto; }, 0),

    /* ═══════════════════════════════════════════════════════════════════════════
       PARCHE PT — Bloque 15:  PRELACIÓN DE LA INFORMACIÓN DEL AÑO EN CURSO
    
       La documentación del período anterior sirve de base, pero las cosas
       cambian de un año a otro: el representante legal, el objeto social, la
       estructura del grupo, las cifras. Cuando un mismo dato aparece en el
       informe anterior y en la información cargada para el año gravable en
       curso, prevalece esta última, y la discrepancia queda registrada.
    
       Orden de prelación:
         3 · lo que escribe el usuario
         2 · la información cargada para el año en curso (RUT, cámara de
             comercio, estados financieros, plantilla de operaciones, soportes)
         1 · la documentación del período anterior
       ═════════════════════════════════════════════════════════════════════════ */
    (function () {
      'use strict';

      window.PT_ORIGENES = window.PT_ORIGENES || {};
      window.PT_CONTRADICCIONES = window.PT_CONTRADICCIONES || [];

      var PRIORIDAD = { anterior: 1, ingesta: 2, usuario: 3 };
      var ETIQUETA = {
        anterior: 'documentación del período anterior',
        ingesta: 'información cargada para el año en curso',
        usuario: 'diligenciado por el usuario'
      };

      function normaliza(v) {
        return String(v === null || v === undefined ? '' : v).replace(/\s+/g, ' ').trim().toLowerCase();
      }
      function esNumero(v) { return /^[\d.,\s$-]+$/.test(String(v || '')) && /\d/.test(String(v || '')); }
      function aNum(v) { var n = parseFloat(String(v).replace(/[^0-9.\-]/g, '')); return isFinite(n) ? n : null; }

      function sonEquivalentes(v1, v2, campo) {
        var n1 = normaliza(v1), n2 = normaliza(v2);
        if (n1 === n2) return true;
        if (!n1 || !n2) return false;

        var esDesarrolloOEncargo = function (s) {
          return /programaci[oó]n|desarrollo|encargo|contrato/i.test(s);
        };
        if (esDesarrolloOEncargo(n1) && esDesarrolloOEncargo(n2)) return true;

        if (campo === 'oper' || campo === 'objeto' || campo === 'vinc' || campo === 'ec_prod' || campo === 'ec_vision' || campo === 'actividad' || campo === 'keywords') {
          var sinEspeciales = function (s) {
            return s.toLowerCase()
              .normalize("NFD").replace(/[\u0300-\u036f]/g, "")
              .replace(/\b(contrato|de|del|para|por|encargo|prestacion|prestación|venta|compra|provision|provisión|ingreso|egreso|servicio|servicios|actividad|desarrollo|empresa|realizar|software)\b/g, '')
              .replace(/[^a-z0-9]/g, '');
          };
          var k1 = sinEspeciales(n1), k2 = sinEspeciales(n2);
          if (k1 && k2 && (k1 === k2 || k1.indexOf(k2) !== -1 || k2.indexOf(k1) !== -1)) return true;

          var esServicio = function (s) { return /servicio|serv|asesor|consultor|soporte|desarrollo|software|tecnolog/i.test(s); };
          if (esServicio(n1) && esServicio(n2)) return true;

          var esInventario = function (s) { return /inventario|mercader|producto|bien|compra|reventa|distrib/i.test(s); };
          if (esInventario(n1) && esInventario(n2)) return true;

          var esInteres = function (s) { return /inter[eé]s|financ|pr[eé]stamo|mutuo/i.test(s); };
          if (esInteres(n1) && esInteres(n2)) return true;
        }

        if (esNumero(v1) && esNumero(v2)) {
          var a = aNum(v1), b = aNum(v2);
          if (a !== null && b !== null) {
            if (Math.abs(a - b) < 2) return true;
            if (a !== 0 && Math.abs((b - a) / a) < 0.005) return true;
          }
        }

        return false;
      }

      /* Fija el valor de un campo respetando la prelación y dejando constancia
         de cualquier contradicción. */
      window.ptFijarCampo = function (id, valor, fuente, origen) {
        var e = $(id);
        if (!e || valor === null || valor === undefined || String(valor).trim() === '') return false;
        origen = origen || 'ingesta';
        var prio = PRIORIDAD[origen] || 2;
        var previo = window.PT_ORIGENES[id];
        var actual = String(e.value || '').trim();

        /* si el campo tiene contenido que no registramos, se atribuye al usuario */
        if (actual && !previo) {
          previo = { valor: actual, fuente: 'diligenciado en el formulario', origen: 'usuario', prioridad: 3 };
          window.PT_ORIGENES[id] = previo;
        }

        if (previo && !sonEquivalentes(previo.valor, valor, id)) {
          var difNum = null;
          if (esNumero(previo.valor) && esNumero(valor)) {
            var a = aNum(previo.valor), b = aNum(valor);
            if (a !== null && b !== null && a !== 0) difNum = ((b - a) / Math.abs(a)) * 100;
          }
          window.PT_CONTRADICCIONES.push({
            campo: id,
            valorPrevio: String(previo.valor).slice(0, 300), origenPrevio: previo.origen, fuentePrevia: previo.fuente,
            valorNuevo: String(valor).slice(0, 300), origenNuevo: origen, fuenteNueva: fuente || ETIQUETA[origen],
            variacion: difNum,
            prevalece: prio >= (previo.prioridad || 1) ? origen : previo.origen,
            fecha: new Date().toISOString()
          });
        }

        if (!previo || prio >= previo.prioridad) {
          e.value = valor;
          window.PT_ORIGENES[id] = { valor: String(valor), fuente: fuente || ETIQUETA[origen], origen: origen, prioridad: prio };
          try { ptPintarContradicciones(); } catch (x) { }
          return true;
        }
        try { ptPintarContradicciones(); } catch (x) { }
        return false;
      };

      /* ───────────────────────────────────────────────────────────────────────
         ENGANCHE CON LOS MÓDULOS QUE DILIGENCIAN CAMPOS
         ─────────────────────────────────────────────────────────────────────── */

      /* 1 · lo que viene del informe anterior entra con la menor prelación */
      if (typeof window.ptDistribuir === 'function') {
        var _distPrev = window.ptDistribuir;
        window.ptDistribuir = function (texto, fuente) {
          var antes = {};
          ['ec_vision', 'ec_prod', 'ec_ingr', 'ec_reparto', 'ec_geo', 'ec_valor', 'ec_fin', 'ec_intang', 'ec_mercado', 'ec_estrategia']
            .forEach(function (id) { var e = $(id); antes[id] = e ? e.value : ''; });
          var r = _distPrev(texto, fuente);
          Object.keys(antes).forEach(function (id) {
            var e = $(id); if (!e) return;
            if (e.value && e.value !== antes[id]) {
              window.PT_ORIGENES[id] = { valor: e.value, fuente: fuente || 'informe anterior', origen: 'anterior', prioridad: 1 };
            }
          });
          return r;
        };
      }

      if (typeof window.ptIdentidadDesdeAnterior === 'function') {

        /* bloques de código y JSON crudo */

        t = t.replace(/\{\s*"[a-zA-Z_]+"\s*:[\s\S]{0,4000}?\}\s*$/g, ' ');
        /* preámbulos conversacionales, aplicados repetidamente al inicio */
        var antes;
        do { antes = t; PREAMBULOS.forEach(function (rx) { t = t.replace(rx, ''); }); } while (t !== antes);
        /* frases conversacionales sueltas dentro del texto */

        return t.replace(/[ \t]{2,}/g, ' ').replace(/\n{3,}/g, '\n\n').trim();

      /* ── detección de líneas y bloques que son tabla de contenido ─────────── */
      function esLineaIndice(l) {
        l = String(l || '');
        /* «Título .......... 12»  o  «Título   12»  con puntos guía */
        if (/.{3,}[.\s]{4,}\d{1,3}\s*$/.test(l) && (l.match(/\./g) || []).length >= 3) return true;
        /* numeración de índice: «4. Análisis …… 12» */
        if (/^\s*(\d+(\.\d+)*|[A-Z]|[IVX]+)[.\)]\s+.{3,60}\s+\d{1,3}\s*$/.test(l)) return true;
        return false;
      }
      window.ptEsLineaIndice = esLineaIndice;

      function esBloqueIndice(texto) {
        var lineas = String(texto).split(/\n/).map(function (x) { return x.trim(); }).filter(function (x) { return x; });
        if (!lineas.length) return false;
        var idx = lineas.filter(esLineaIndice).length;
        /* si la mitad o más de las líneas parecen entradas de índice, es TOC */
        return idx >= 2 && idx / lineas.length >= 0.4;
      }
      window.ptEsBloqueIndice = esBloqueIndice;

      /* limpia fragmentos de índice de un texto conservando la prosa real */
      function limpiarIndice(texto) {
        if (!texto) return texto;
        var lineas = String(texto).split(/\n/);
        var utiles = lineas.filter(function (l) { return l.trim() && !esLineaIndice(l); });
        var t = utiles.join('\n');
        /* fragmentos de índice en línea: «Competencia ...... 12 Riesgos ...... 13» */
        t = t.replace(/[^.\n]{3,40}[.\s]{4,}\d{1,3}(\s+[^.\n]{3,40}[.\s]{4,}\d{1,3})+/g, ' ');
        return t.replace(/[ \t]{2,}/g, ' ').replace(/\n{3,}/g, '\n\n').trim();

      function claveNombre(s) {
        return String(s || '').toUpperCase()

      function annexLegislacionF() {
        return '<h2 class="rep">Anexo F — Legislación colombiana en materia de precios de transferencia</h2>' +
          '<p>El Congreso de la República, mediante las Leyes 788 de 2002 y 863 de 2003, incorporó el régimen de precios de transferencia a la legislación colombiana, con aplicación a partir del año gravable 2004. La Ley 1607 de 2012 (arts. 111 a 121, Capítulo VI — Normas Antievasión) modificó integralmente los artículos 260-1 a 260-11 del Estatuto Tributario, reglamentados por el Decreto 1625 de 2016 y, en particular, por el Decreto 2120 del 15 de diciembre de 2017, que alineó la documentación comprobatoria al estándar BEPS Acción 13 (Informe Local, Informe Maestro e Informe País por País). Las Directrices de la OCDE se aplican como criterio auxiliar de interpretación, no vinculante.</p>' +

          '<h3 class="rep">F.1 Vinculación económica y partes relacionadas (Art. 260-1 E.T.)</h3>' +
          '<p>El régimen se aplica a los contribuyentes del impuesto sobre la renta que celebren operaciones con vinculados económicos o partes relacionadas domiciliadas o residentes en el exterior, en zona franca, o con quienes se presuma vinculación por ubicarse en jurisdicciones no cooperantes. El artículo 260-1 define la vinculación, entre otros, por: subordinación o control (cuando el poder de decisión de una entidad está sometido a la voluntad de otra, que será su matriz o controlante); sucursales y agencias; establecimientos permanentes; vinculación por capital, por administración o por participación conjunta en otras sociedades; y consorcios o uniones temporales con sus miembros. La vinculación económica <b>subsiste</b> aun cuando la operación se realice por medio de terceros no vinculados interpuestos entre las partes relacionadas. Las compañías utilizadas como comparables deben ser <b>independientes</b>: se excluyen las entidades controladas, subordinadas o que a su vez tengan vinculación con la parte examinada o con el grupo económico.</p>' +

          '<h3 class="rep">F.2 Principio de plena competencia (Art. 260-2 E.T.)</h3>' +
          '<p>Los contribuyentes que celebren operaciones con vinculados del exterior, en zonas francas o con residentes en jurisdicciones no cooperantes deben determinar, para efectos del impuesto sobre la renta y complementarios, sus ingresos ordinarios y extraordinarios y sus costos y deducciones, considerando para esas operaciones los precios y márgenes de utilidad que hubieran pactado partes independientes en operaciones comparables (principio de plena competencia). El parágrafo del artículo dispone que las normas de precios de transferencia producen efectos <b>únicamente</b> en la determinación del impuesto sobre la renta y complementarios, salvo las excepciones que la propia norma señale; por esa razón, los ajustes de precios de transferencia no se trasladan automáticamente a la determinación del IVA.</p>' +

          '<h3 class="rep">F.3 Métodos para determinar el precio o margen de utilidad (Art. 260-3 E.T.)</h3>' +
          '<p>El precio o margen de utilidad en las operaciones entre vinculados podrá determinarse mediante la aplicación individual o combinada de cualquiera de los siguientes cinco métodos, según cuál resulte más apropiado en función de las características de la operación evaluada. Para operaciones de <i>commodities</i> solo procede acudir a un método distinto del Precio Comparable no Controlado en casos excepcionales, siempre que la documentación comprobatoria incluya las razones económicas, financieras y técnicas que lo justifiquen.</p>' +

          '<h4 class="rep">F.3.1 Precio Comparable no Controlado (PC)</h4>' +
          '<p>Consiste en valorar el precio del bien o servicio en una operación entre vinculados al precio de un bien o servicio idéntico o de características similares en una operación entre terceros independientes en circunstancias comparables, efectuando los ajustes necesarios para obtener la equivalencia. Es, en general, la medida más confiable cuando las transacciones son idénticas o solo presentan diferencias menores fácilmente cuantificables, pero exige un alto nivel de comparabilidad de producto y de funciones. Los ajustes más habituales corresponden a diferencias en calidad del producto, términos contractuales, nivel de mercado, mercados geográficos, fecha de la transacción, propiedad intangible asociada y riesgo de moneda extranjera.</p>' +

          '<h4 class="rep">F.3.2 Precio de Reventa (PR)</h4>' +
          '<p>Determina el precio de compra de un bien o de prestación de un servicio entre vinculados, multiplicando el precio de reventa del bien o servicio a partes independientes por el resultado de disminuir de la unidad el margen bruto de utilidad obtenido entre partes independientes en operaciones comparables: <b>PR = P × (1 − MB)</b>, donde P es el precio de reventa acordado con un tercero independiente y MB (utilidad bruta / ventas netas) es el margen bruto de referencia. El punto de partida es el precio al que un producto comprado a un vinculado se revende a un tercero; el margen bruto adecuado puede obtenerse de operaciones comparables del mismo revendedor (interno) o de otros revendedores en operaciones de igual naturaleza (externo). Se utiliza sobre todo en distribuidores que revenden productos sin transformarlos físicamente ni agregarles valor relevante, y requiere ajustes por rotación de inventarios, términos contractuales y costos de transporte, entre otros.</p>' +

          '<h4 class="rep">F.3.3 Costo Adicionado (CA)</h4>' +
          '<p>Consiste en multiplicar el costo de los bienes o servicios por el resultado de sumar a la unidad el margen de utilidad bruta obtenido entre partes independientes en operaciones comparables. El punto de partida es el costo asumido por el proveedor en la operación con el vinculado; el precio de venta de plena competencia se obtiene añadiendo a ese costo el margen de ganancia adecuado, considerando las funciones realizadas, los riesgos incurridos, los activos intangibles empleados y la complejidad de la fabricación. El margen de referencia puede derivarse de operaciones del mismo proveedor con terceros (interno) o de otros proveedores comparables (externo), con ajustes por rotación de inventarios, términos contractuales y costos de transporte.</p>' +

          '<h4 class="rep">F.3.4 Márgenes Transaccionales de Utilidad de Operación (TU)</h4>' +
          '<p>Evalúa la utilidad de operación obtenida por el contribuyente en la operación con el vinculado frente a una base pertinente (ventas, costos o activos), comparándola con la utilidad de operación de compañías independientes en operaciones comparables. Su confiabilidad puede verse afectada por diferencias en la estructura de costos, en la experiencia del negocio o en la eficiencia administrativa, y suele requerir ajustes por clasificaciones contables, condiciones de crédito, inventario, riesgo de divisas, mercados geográficos y circunstancias del negocio. Los indicadores de rentabilidad (PLI) habitualmente empleados bajo este método son:</p>' +
          '<ul>' +
          '<li><b>Margen Operacional (MO)</b> = Utilidad operativa / Ventas netas. Mide la rentabilidad de la actividad principal de la compañía y es menos sensible a la distribución de gastos entre costo de ventas y gastos operativos; se emplea sobre todo en manufactura y distribución.</li>' +
          '<li><b>Margen sobre Costos y Gastos (MCG)</b> = Utilidad operativa / (Costo de ventas + Gastos operativos). Apropiado en prestación de servicios, ensamble y manufactura, y sensible al uso del capital.</li>' +
          '<li><b>Rendimiento sobre Activos (ROA)</b> = Utilidad operativa / Activo total. Apropiado cuando la rentabilidad de la compañía depende principalmente del uso de sus activos; tiene menor sensibilidad a las diferencias de clasificación contable entre gastos operativos y costo de ventas y exige menor comparabilidad funcional, pero depende de la información del balance general y de sus criterios de clasificación.</li>' +
          '<li><b>Razón Berry</b> = Utilidad bruta / Gastos de operación. Es sensible a las diferencias de clasificación contable entre gastos operativos y costo de ventas, por lo que se reserva para casos en que esa clasificación es homogénea entre la parte examinada y las comparables.</li>' +
          '</ul>' +

          '<h4 class="rep">F.3.5 Partición de Utilidades (PU)</h4>' +
          '<p>Asigna la utilidad o pérdida de operación global de la transacción entre las partes relacionadas, en proporción a la contribución relativa de cada una (volumen de activos, costos y gastos asumidos). Evalúa si esa asignación es compatible con el principio de plena competencia, con base en el valor relativo de la contribución de cada parte. Se aplica sobre todo a operaciones altamente integradas que no pueden evaluarse por separado y no requiere necesariamente comparaciones con operaciones de terceros independientes; en contrapartida, su confiabilidad es especialmente sensible a la calidad de los datos y de los supuestos utilizados.</p>' +

          '<p>Se entiende que las operaciones son comparables cuando no existen diferencias entre sus características económicas relevantes que puedan afectar de forma significativa el precio, la contraprestación o el margen de utilidad, o cuando tales diferencias pueden eliminarse mediante ajustes técnicos razonables. Para establecer la comparabilidad se deben considerar los siguientes cinco factores.</p>' +

          '<li><b>Uso o enajenación de bienes tangibles:</b> características físicas, calidad, confiabilidad, disponibilidad del bien y volumen de la oferta.</li>' +
          '<li><b>Explotación o transferencia de intangibles:</b> clase de bien (patente, marca, nombre comercial o know-how), duración y grado de protección, y beneficios esperados de su uso.</li>' +
          '<li><b>Enajenación de acciones:</b> valor presente de las utilidades o flujos de efectivo proyectados, o la cotización bursátil del emisor al último día de la enajenación.</li>' +
          '</ul>' +
          '<h4 class="rep">F.4.2 Funciones o actividades económicas</h4>' +
          '<p>Las funciones o actividades económicas desarrolladas por cada parte, incluyendo los activos utilizados y los riesgos asumidos en la operación (análisis funcional).</p>' +
          '<h4 class="rep">F.4.3 Términos contractuales</h4>' +
