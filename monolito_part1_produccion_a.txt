    function extraerJSONDeRespuestaIA(texto) {
      const s = String(texto || '');
      const inicio = s.indexOf('{');
      if (inicio < 0) throw new Error('La respuesta no contiene un objeto JSON.');
      let profundidad = 0, enCadena = false, escapado = false;
      for (let i = inicio; i < s.length; i++) {
        const c = s[i];
        if (enCadena) {
          if (escapado) escapado = false;
          else if (c === '\\') escapado = true;
          else if (c === '"') enCadena = false;
          continue;
        }
        if (c === '"') { enCadena = true; continue; }
        if (c === '{') profundidad++;
        else if (c === '}') { profundidad--; if (profundidad === 0) return JSON.parse(s.slice(inicio, i + 1)); }
      }

    /* Compara NITs tolerando el dígito de verificación (DV) pegado sin separador:

    function nitCoincide(a, b) {
      a = _digits(a); b = _digits(b);
      if (!a || !b) return true;
      if (a === b) return true;
      if (a.length === b.length + 1) return a.slice(0, -1) === b;
      if (b.length === a.length + 1) return b.slice(0, -1) === a;
      return false;
    }

    /* Reconstruye el texto de una página usando la posición real (x,y) de cada
       fragmento, en vez del orden del content-stream del PDF (que en formularios
       oficiales suele venir revuelto entre campos, sin relación con el orden de
       lectura visual). Agrupa fragmentos en filas por coordenada Y y los ordena
       por X dentro de cada fila. Un hueco horizontal pequeño (hasta ~2x la altura
       del texto) se concatena sin separador — así se reconstruyen números que el
       formulario reparte en casillas contiguas ("9 0 0 1…" → "9001…"); un hueco
       mayor se trata como separación real entre palabras o campos distintos. */
    function pdfExtraerTextoPosicional(items) {
      const filas = new Map();
      for (const it of items) {
        if (!it.str || !it.str.trim()) continue;
        const y = Math.round(it.transform[5]);
        let clave = null;
        for (const k of filas.keys()) { if (Math.abs(k - y) <= 2) { clave = k; break; } }
        if (clave === null) clave = y;
        if (!filas.has(clave)) filas.set(clave, []);
        filas.get(clave).push({ x: it.transform[4], w: it.width, h: it.height || Math.abs(it.transform[3]) || 8, str: it.str });
      }
      const filasOrdenadas = [...filas.entries()].sort((a, b) => b[0] - a[0]); // Y descendente = arriba a abajo
      const lineas = [];
      for (const [, arr] of filasOrdenadas) {
        arr.sort((a, b) => a.x - b.x);
        let linea = '', prev = null;
        for (const it of arr) {
          if (prev) {
            const gap = it.x - (prev.x + prev.w), h = it.h;
            if (gap > h * 4) linea += '  ';
            else if (gap > h * 2) linea += ' ';
            // si no (gap<=h*2): sin separador
          }
          linea += it.str; prev = it;
        }
        lineas.push(linea);
      }
      return lineas.join('\n');
    }

    function parseRUT(txt) {
      const t = String(txt || '').replace(/\s+/g, ' '); const o = {}; let m; const NITP = '((?<![0-9])(?:[0-9]{1,3}(?:\\.[0-9]{3}){1,3}|[0-9]{9,10})(?![0-9]))';
      if (m = t.match(new RegExp('n[uú]mero de identificaci[oó]n tributaria\\s*\\(?nit\\)?[^0-9]{0,12}' + NITP, 'i'))) { o.nit = _digits(m[1]); }
      else if (m = t.match(new RegExp('\\bNIT\\b[^0-9]{0,8}' + NITP, 'i'))) { o.nit = _digits(m[1]); }
      else if (m = t.match(new RegExp(NITP))) { o.nit = _digits(m[1]); }
      /* el RUT trae el DV (dígito de verificación) en la columna contigua a la del
         NIT ("5. Número de Identificación Tributaria (NIT)  6. DV"); al reconstruir
         el texto por posición ambos quedan pegados en un solo número de 10 dígitos.
         El NIT colombiano de persona jurídica tiene 9 dígitos: se descarta el DV. */
      if (o.nit && o.nit.length === 10) o.nit = o.nit.slice(0, 9);
      if (m = t.match(/raz[oó]n social[:\s]*([A-ZÁÉÍÓÚÑ0-9][^\n]{2,80}?)(?=\s+(?:3[5-9]|4[0-9]|nombre|sigla|direcci|tipo de contribuyente|primer apellido|actividad|responsab|representaci|c[oó]digo)|$)/i)) o.ent = _tidy(m[1]);
      if (m = t.match(/actividad(?:\s+econ[oó]mica)?(?:\s+principal)?[^0-9]{0,20}(?:c[oó]digo[^0-9]{0,5})?([0-9]{4})\b/i)) o.ciiu = m[1];
      if (m = t.match(/direcci[oó]n(?:\s+principal)?[:\s]*([^\n]{5,80}?)(?=\s+(?:departamento|municipio|ciudad|correo|tel[eé]fono|$))/i)) o.dir = _tidy(m[1]);
      if (m = t.match(/representaci[oó]n legal[:\s]*([A-ZÁÉÍÓÚÑ][^\n]{3,60})/i)) o.replegal = _tidy(m[1]);
      return o;
    }

    /* Detecta la fila de encabezados entre las primeras filas de un Excel/CSV
       importado: algunos exportes (p. ej. Capital IQ) traen filas de título
       o vacías antes de los encabezados reales. Se elige la fila con más

    function findHeaderRow(rows) {
      let mejorFila = 0, mejorPuntaje = -1;
      const tope = Math.min(rows.length, 15);
      for (let i = 0; i < tope; i++) {
        const celdas = (rows[i] || []).filter(x => String(x == null ? '' : x).trim() !== '');
        if (!celdas.length) continue;
        const textuales = celdas.filter(x => isNaN(parseFloat(x)) || !isFinite(x));
        const puntaje = textuales.length + (celdas.length >= 3 ? 1 : 0);
        if (puntaje > mejorPuntaje) { mejorPuntaje = puntaje; mejorFila = i; }
      }
      return mejorFila;
    }

    async function adjPDFaImagenes(file, tipo, fin) {
      try {

        if (!window.pdfjsLib) { moToast('Motor PDF no disponible: convierta el PDF a imágenes (JPG/PNG) y cárguelas', 'red'); fin(); return; }

    /* ─── Transcribión inteligente para tipo "otros" ─── *
     *  Ruta 1 (pdfjs): extrae texto nativo página a página con pdfExtraerTextoPosicional()
     *  Ruta 2 (Claude OCR): si una página tiene < UMBRAL_CHARS caracteres útiles,
     *    se renderiza a canvas y se envía a Claude para que la transcriba.
     *  El resultado se guarda como { tipo:'texto', contenido, ... } — no como imagen.
     */
    async function adjPDFTranscribir(file, tipo, fin) {
      const UMBRAL_CHARS = 30; // caracteres útiles mínimos por página para considerar nativa
      try {
        if (!window.pdfjsLib) { try { await loadScript('vendor/pdfjs/pdf.min.js'); } catch (e) { } }
        if (!window.pdfjsLib) { moToast('Motor PDF no disponible: convierta el PDF a texto y péguelo manualmente', 'red'); fin(); return; }
        if (window.pdfjsLib.GlobalWorkerOptions && !window.pdfjsLib.GlobalWorkerOptions.workerSrc)
          window.pdfjsLib.GlobalWorkerOptions.workerSrc = 'vendor/pdfjs/pdf.worker.min.js';

        const barra = document.createElement('div');
        barra.style.cssText = 'position:fixed;bottom:16px;left:16px;z-index:1000;background:var(--ink);color:#EDF2F7;padding:10px 14px;border-radius:8px;font-size:12.5px;min-width:290px;box-shadow:var(--shadow)';
        barra.setAttribute('role', 'status'); barra.setAttribute('aria-live', 'polite');
        document.body.appendChild(barra);
        const upd = (i, t, modo) => {
          barra.innerHTML =
            'Transcribiendo «' + escH(file.name) + '»: pág. ' + i + ' de ' + t
            + (modo === 'ocr' ? ' <span style="color:#F0C060">▶ Claude OCR</span>' : ' <span style="color:#7CC4A4">▶ texto nativo</span>')
            + '<div style="height:6px;background:#2A3F4D;border-radius:4px;margin-top:6px;overflow:hidden"><div style="height:100%;width:' + Math.round(i / t * 100) + '%;background:#7CC4A4"></div></div>';
        };
        upd(0, 1, 'nativo');

        const buf = await file.arrayBuffer();
        const pdf = await pdfjsLib.getDocument({ data: buf }).promise;
        const total = pdf.numPages;
        let textoCompleto = '';
        let paginasOCR = 0;

        for (let i = 1; i <= total; i++) {
          const page = await pdf.getPage(i);

          /* Intentar extracción nativa */
          const content = await page.getTextContent();
          const textoNativo = pdfExtraerTextoPosicional(content.items);
          const charsUtiles = textoNativo.replace(/\s+/g, '').length;

          if (charsUtiles >= UMBRAL_CHARS) {
            upd(i, total, 'nativo');
            textoCompleto += '\n--- Página ' + i + ' ---\n' + textoNativo + '\n';
          } else {
            /* Fallback: renderizar página a canvas y enviar a Claude */
            upd(i, total, 'ocr');
            const vp = page.getViewport({ scale: 2.0 });

    var RX_DIACRITICOS = new RegExp('[' + String.fromCharCode(768) + '-' + String.fromCharCode(879) + ']', 'g');
    function _ptTokensSignificativos(frase) {
      return String(frase || '').toLowerCase()
        .normalize('NFD').replace(RX_DIACRITICOS, '')
        .replace(/[^a-z0-9\s]/g, ' ')
        .split(/\s+/)

    function moActMatch(o, act, aiRec) {

      const fraseIngles = (typeof ptTraducirKeyword === 'function') ? ptTraducirKeyword(act.resumen) : act.resumen;
      const tokens = _ptTokensSignificativos(fraseIngles).concat(_ptTokensSignificativos(act.resumen));
      let hits = 0;
      tokens.forEach(function (t) { if (base.indexOf(t) >= 0) hits++; });
      const posibles = tokens.length;
      return { factor: Math.max(0.15, Math.min(1, hits / 4)), hits: hits, posibles: posibles, hayAct: true, tieneDesc: !!descText.trim() };

    function _ptChunk(arr, size) {
      const out = [];
      for (let i = 0; i < arr.length; i += size) out.push(arr.slice(i, i + size));
      return out;
    }
    async function _ptPool(items, worker, limite) {
      const resultados = new Array(items.length);
      let idx = 0;
      async function siguiente() {
        while (idx < items.length) {
          const mio = idx++;
          resultados[mio] = await worker(items[mio], mio);
        }
      }
      const corredores = Array.from({ length: Math.min(limite, items.length) }, siguiente);
      await Promise.all(corredores);
      return resultados;
    }

    /* ── A1 · quart(): ambas opciones cumplen el Art. 1.2.2.2.4.5 ──
       Verificado contra el decreto (normograma DIAN): posiciones (base 1)
       Mediana=(n+1)/2 · P25=(Med+1)/2 · P75=Med−1+P25 ≡ p*(n−1) en base 0.
       La fórmula p*(n+1)−1 del modo 'dian' original NO cumple (desfase 0,5
       posición ∀n, demostrado algebraicamente). Se elimina y se deja constancia. */
    function quart(sorted, p, method) {
      const n = sorted.length; if (!n) return null; if (n === 1) return sorted[0];
      let pos = p * (n - 1);                       // única fórmula conforme al decreto
      pos = Math.max(0, Math.min(n - 1, pos));
      const lo = Math.floor(pos), hi = Math.min(lo + 1, n - 1);
      return sorted[lo] + (sorted[hi] - sorted[lo]) * (pos - lo);
    }

    /* ── A3 · adjustInfo(): egreso>0 para activar tope; fundamento corregido ──
       Tope: Concepto DIAN 010813 int 1212 del 19-dic-2024 (verificado):
       «Los ajustes en costos y/o deducciones no pueden exceder el valor de las
       operaciones de egreso del contribuyente bajo estudio, y no se podrán
       aplicar ajustes proporcionales.»
       Sentido del ajuste: el Art. 260-2 E.T. NO contiene prohibición expresa de
       disminuir la base (verificado contra el texto íntegro). El criterio de no
       practicar ajustes que reduzcan la base se funda en la doctrina DIAN
       (Concepto 6610/2026: el ajuste procede «aumentando los ingresos o
       disminuyendo los costos y deducciones») y en la ausencia de autorización
       legal para el sentido contrario. Se reporta como criterio doctrinal. */
    function adjustInfo(T, tPLI, st, base, unitMult) {
      if (!st || tPLI === null) return null;
      const within = tPLI >= st.p25 && tPLI <= st.p75;
      if (within) return { within: true, raw: 0, capped: 0, flag: false };
      let raw = (st.med - tPLI) * base * unitMult;
      const egreso = num($('egreso').value);
      let capped = raw, flag = false;
      if (egreso !== null && egreso > 0 && Math.abs(raw) > egreso) { capped = Math.sign(raw) * egreso; flag = true; } // A3: exige >0
      let improcedente = false;
      if (raw < 0) { capped = 0; flag = false; improcedente = true; } // criterio doctrinal (Conc. 6610/2026); no cita legal expresa
      return { within: false, raw, capped, flag, improcedente, dir: tPLI < st.p25 ? 'por debajo' : 'por encima' };
    }

      const texto = blocks.filter(b => b.type === 'text').map(b => b.text).join('\n').trim();
      blocks.filter(b => b.type === 'web_search_tool_result').forEach(b => {
        (b.content || []).forEach(r => { if (r && r.url) fuentes.push({ t: r.title || r.url, u: r.url }); });
      }); // block.content, no .results
      return { texto, fuentes, busco: fuentes.length > 0 };
    }
