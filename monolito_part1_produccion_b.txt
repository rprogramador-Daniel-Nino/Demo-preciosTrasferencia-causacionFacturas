    /**
     * AUTO-LIMPIAR DATOS VIEJOS
     */
    async function autoLimpiarDatosViejos() {
      try {
        let allKeys = [];

        if (typeof stList === 'function') {
          allKeys = await stList('pt:study:');
        } else if (typeof localStorage !== 'undefined') {
          allKeys = Object.keys(localStorage).filter(k => k.startsWith('pt:study:'));
        }

        if (allKeys.length <= autoSaveConfig.keepOldStudies) {
          console.log(`✅ Almacenamiento limpio (${allKeys.length} estudios)`);
          return;
        }

        const studies = [];
        for (const key of allKeys) {
          let data = null;

          if (typeof stGet === 'function') {
            data = await stGet(key);
          } else if (typeof localStorage !== 'undefined') {
            data = localStorage.getItem(key);
          }

          if (data) {
            try {
              const parsed = typeof data === 'string' ? JSON.parse(data) : data;
              studies.push({
                key,
                lastModified: parsed.lastModified || Date.now(),
                size: JSON.stringify(data).length
              });
            } catch (e) { }
          }
        }

        studies.sort((a, b) => b.lastModified - a.lastModified);

        const toDelete = studies.slice(autoSaveConfig.keepOldStudies);
        for (const study of toDelete) {
          if (typeof stDel === 'function') {
            await stDel(study.key);
          } else if (typeof localStorage !== 'undefined') {
            localStorage.removeItem(study.key);
          }
        }

        console.log(`🧹 Auto-limpieza: se eliminaron ${toDelete.length} estudios antiguos`);
        autoSaveConfig.lastClean = Date.now();

      } catch (error) {
        console.error('Error en auto-limpieza:', error);
      }
    }

    // MÓDULO V3.5 — PERSISTENCIA IndexedDB + BASE AÑO ANTERIOR + DRAFTS
    // ADITIVO: no modifica ninguna función previa. Reutiliza readForm/stSet/ADJ.
    // ═══════════════════════════════════════════════════════════════════════════
    var PT35 = (function () {
      var db = null, DBNAME = 'PT35', STORE = 'estudios', listo = null;
      function ready() { if (db) return Promise.resolve(db); if (!listo) listo = abrir(); return listo; }

      function abrir() {
        return new Promise(function (res, rej) {
          var r = indexedDB.open(DBNAME, 1);
          r.onupgradeneeded = function (e) {
            var d = e.target.result;
            if (!d.objectStoreNames.contains(STORE)) d.createObjectStore(STORE, { keyPath: 'id' });
          };
          r.onsuccess = function () { db = r.result; res(db); };
          r.onerror = function () { rej(r.error); };
        });
      }

      function tx(mode) { return db.transaction([STORE], mode).objectStore(STORE); }
      function put(o) {
        return new Promise(function (res, rej) {
          var q = tx('readwrite').put(o);
          q.onsuccess = function () { res(o) }; q.onerror = function () { rej(q.error) };
        });
      }
      function all() {
        return new Promise(function (res, rej) {
          var q = tx('readonly').getAll();
          q.onsuccess = function () { res(q.result || []) }; q.onerror = function () { rej(q.error) };
        });
      }
      function del(id) {
        return new Promise(function (res) {
          var q = tx('readwrite').delete(id);
          q.onsuccess = function () { res() }; q.onerror = function () { res() };
        });
      }

      function idDe(tipo) {
        var e = (document.getElementById('ent') && document.getElementById('ent').value || 'estudio')
          .toLowerCase().replace(/[^a-z0-9]+/g, '_');
        var y = (document.getElementById('histyear') && document.getElementById('histyear').value) ||
          (document.getElementById('anio') && document.getElementById('anio').value) || 'sa';
        return tipo + '__' + e + '__' + y;
      }

      var PESOS = { eeff: 25, operaciones: 35, comparables: 25, analisis: 15 };
      function progreso(d) { var t = 0; for (var k in PESOS) { if (d && d[k]) t += PESOS[k]; } return t; }

      // ---- snapshot del formulario COMPLETO, usando la función nativa del sistema ----
      async function snapshot() {
        var form = (typeof readForm === 'function') ? readForm() : {};
        var autoOps = null;
        if (typeof stGet === 'function') { try { autoOps = await stGet('autoOps'); } catch (e) { } }
        return {
          eeff: (form.patrimonio || form.ingbrutos) ? { patrimonio: form.patrimonio, ingbrutos: form.ingbrutos } : null,
          operaciones: autoOps || (form.monto ? [{ monto: form.monto }] : null),
          comparables: (form.comparables && form.comparables.length) ? form.comparables : null,
          analisis: form.conclusion || null,
          form: form
        };
      }

      function guardar(tipo) {
        return ready().then(snapshot).then(function (d) {
          var reg = {
            id: idDe(tipo || 'actual'), tipo: tipo || 'actual',
            ent: (document.getElementById('ent') || {}).value || '',
            anio: (document.getElementById('histyear') || {}).value || '',
            datos: d, progreso: progreso(d), ts: new Date().toISOString()
          };
          return put(reg);
        });
      }

      // ---- restaurar SIN pisar el formulario si ya tiene datos ----
      function restaurar() {
        return ready().then(all).then(function (regs) {
          if (!regs.length) { log('Sin trabajo previo guardado.'); return null; }
          var act = regs.filter(function (r) { return r.tipo !== 'anterior'; })
            .sort(function (a, b) { return new Date(b.ts) - new Date(a.ts); })[0];
          if (act && act.datos && act.datos.form && typeof fillForm === 'function') {
            try { fillForm(act.datos.form); } catch (e) { }
          } else if (act && act.datos && act.datos.form) {
            // relleno genérico campo a campo si no existe fillForm
            Object.keys(act.datos.form).forEach(function (k) {
              var el = document.getElementById(k);
              if (el && (el.value === '' || el.value == null)) el.value = act.datos.form[k];
            });
          }
          if (act) log('♻️ Sesión restaurada: ' + (act.ent || 'estudio') + ' — ' + act.progreso + '%');
          return act;
        });
      }

      function base() {
        return ready().then(all).then(function (r) { return r.filter(function (x) { return x.tipo === 'anterior'; })[0] || null; });
      }

      function cargarAnterior(file) {
        return new Promise(function (res, rej) {
          var fr = new FileReader();
          fr.onload = async function () {
            try {
              var reg;
              if (/\.json$/i.test(file.name)) {
                var j = JSON.parse(fr.result);
                reg = {
                  id: 'anterior__' + ((j.ent || 'x').toLowerCase().replace(/[^a-z0-9]+/g, '_')) + '__' + (j.anio || ''),
                  tipo: 'anterior', ent: j.ent || '', anio: j.anio || '',
                  datos: {
                    eeff: { patrimonio: j.patrimonio, ingbrutos: j.ingbrutos },
                    operaciones: j.operaciones || null, comparables: j.comparables || null,
                    analisis: j.conclusion || null, form: j
                  },
                  progreso: 100, ts: new Date().toISOString()
                };
              } else {
                await loadXLSX();
                var wb = XLSX.read(new Uint8Array(fr.result), { type: 'array' }), hojas = {};
                wb.SheetNames.forEach(function (n) { hojas[n] = XLSX.utils.sheet_to_json(wb.Sheets[n], { header: 1 }); });
                var ops = []; var h = hojas['Op. Vinculados Economicos'] || hojas[wb.SheetNames[0]] || [];
                for (var i = 11; i < h.length; i++) { var f = h[i] || []; if (f[1]) ops.push({ vinculado: f[1], nit: f[2], pais: f[3], tipo_operacion: f[5] }); }
                reg = {
                  id: 'anterior__' + file.name.toLowerCase().replace(/[^a-z0-9]+/g, '_'), tipo: 'anterior',
                  ent: file.name, anio: '', datos: { operaciones: ops, comparables: null, eeff: null, analisis: null },
                  progreso: 100, ts: new Date().toISOString()
                };
              }
              put(reg).then(function () {
                log('📁 Base año anterior cargada: ' + (reg.ent || file.name) +
                  (reg.datos.operaciones ? (' — ' + reg.datos.operaciones.length + ' operaciones') : '')); res(reg);
              });
            } catch (e) { rej(e); }
          };
          fr.onerror = function () { rej(fr.error) };
          if (/\.json$/i.test(file.name)) fr.readAsText(file); else fr.readAsArrayBuffer(file);
        });
      }

      // sugerir comparables del año anterior (no los impone)
      function sugerirComparables() {
        return base().then(function (b) {
          if (!b || !b.datos.comparables || !b.datos.comparables.length) { log('No hay comparables en la base anterior.'); return null; }
          log('💡 ' + b.datos.comparables.length + ' comparables del año anterior disponibles como sugerencia.');
          return b.datos.comparables;
        });
      }

      // borrador progresivo con lo que haya cargado
      async function borrador() {
        var d = await snapshot(), p = progreso(d), sec = [];
        if (d.eeff) sec.push('I. ANÁLISIS FINANCIERO\nPatrimonio: ' + (d.eeff.patrimonio || '—') + '\nIngresos brutos: ' + (d.eeff.ingbrutos || '—'));
        if (d.operaciones) sec.push('II. OPERACIONES CON VINCULADOS\nRegistros: ' + (d.operaciones.length || 0));
        if (d.comparables) sec.push('III. ANÁLISIS DE COMPARABLES\nEmpresas: ' + (d.comparables.length || 0));
        if (d.analisis) sec.push('IV. CONCLUSIÓN\n' + d.analisis);
        if (!sec.length) sec.push('(Sin información cargada todavía — borrador vacío)');
        return {
          progreso: p,
          texto: 'BORRADOR INFORME PT — ' + ((document.getElementById('ent') || {}).value || '') + '\n' +
            'Estado: ' + p + '% completado — ' + (p < 100 ? 'EN CONSTRUCCIÓN' : 'COMPLETO') + '\n' +
            'Generado: ' + new Date().toLocaleString('es-CO') + '\n\n' + sec.join('\n\n')
        };
      }

      async function descargarBorrador() {
        var b = await borrador();
        var a = document.createElement('a');
        a.href = URL.createObjectURL(new Blob([b.texto], { type: 'text/plain;charset=utf-8' }));
        a.download = 'Borrador_PT_' + b.progreso + 'pct.txt'; a.click();
        log('📥 Borrador descargado (' + b.progreso + '%)');
      }

      function log(m) { try { console.log('[PT3.5] ' + m); } catch (e) { } }

      function init() {
        ready().then(function () {
          log('IndexedDB listo.');
          return restaurar();
        }).then(function () {
          setInterval(function () { guardar('actual'); }, 30000);
          document.addEventListener('visibilitychange', function () { if (document.hidden) guardar('actual'); });
          window.addEventListener('beforeunload', function () { try { guardar('actual'); } catch (e) { } });
          log('Autoguardado IndexedDB activo (30s + al salir).');
        }).catch(function (e) { log('Error init: ' + e); });
      }

      return {

    function parseRenta(txt) {
      var t = String(txt || '').replace(/\s+/g, ' ');
      var o = {}; var g = function (re) { var m = t.match(re); return m ? _digits(m[1]) : null; };
      var NUM = '([0-9]{1,3}(?:[.,][0-9]{3})+|[0-9]{6,})';
      // por número de renglón (formato 110) con respaldo por etiqueta
      o.nit = (function () { var m = t.match(/nit[^0-9]{0,6}([0-9]{1,3}(?:\.[0-9]{3}){1,3}|[0-9]{9,10})/i); return m ? _digits(m[1]) : null; })();
      o.anio = (function () { var m = t.match(/a[nñ]o\s*(?:gravable)?[^0-9]{0,4}(20[12][0-9])/i); return m ? m[1] : null; })();
      o.patrimonio = g(new RegExp('(?:patrimonio bruto|total patrimonio bruto|renglon\\s*30)[^0-9]{0,18}' + NUM, 'i'));
      o.ingbrutos = g(new RegExp('(?:total ingresos brutos|ingresos brutos|renglon\\s*4[3-9])[^0-9]{0,18}' + NUM, 'i'));
      o.costos = g(new RegExp('(?:total costos|costos y (?:deducciones|gastos)|renglon\\s*5[0-9])[^0-9]{0,18}' + NUM, 'i'));
      o.rentaliq = g(new RegExp('(?:renta l[ií]quida (?:gravable|ordinaria)|renglon\\s*6[0-9])[^0-9]{0,18}' + NUM, 'i'));
      return o;
    }

    function _pt39num(s) { return (typeof num === 'function') ? num(s) : parseFloat(String(s).replace(/[^0-9.\-]/g, '')); }

    /* ═══ PT3.18 — EXCEL FORMULADO del rango intercuartil (método del decreto) ═══
       Exporta un .xlsx con FÓRMULAS VIVAS (no valores pegados):

       · Parte examinada, posición en el rango y AJUSTE a la mediana formulado.

    function _pt318BuildWB() {
      var kind = ($('pli') && $('pli').value) || 'MO';
      // leer comparables incluidas del set actual
      var comps = [];
      [].slice.call(document.querySelectorAll('#cbody tr')).forEach(function (tr) {
        var ins = [].slice.call(tr.querySelectorAll('input'));
        if (!ins.length) return;
        var name = (ins[0] && ins[0].value || '').trim(); if (!name) return;
        /* estructura real de compRow: ins[0]=nombre, ins[1]=ventas, ins[2]=costo, ins[3]=utilidad op.
           (el «tipo» es un <select>, no input) */
        var s = num(ins[1] && ins[1].value), c = num(ins[2] && ins[2].value), op = num(ins[3] && ins[3].value);
        if (s == null) return;
        comps.push({ name: name, s: s, c: c == null ? 0 : c, op: op == null ? 0 : op });
      });
      var t_s = num($('t_s') && $('t_s').value) || 0, t_c = num($('t_c') && $('t_c').value) || 0, t_op = num($('t_op') && $('t_op').value) || 0;
      var n = comps.length;
      var A = XLSX.utils.book_new();
      var ws = {};
      var setC = function (ref, v, f, fmt) { ws[ref] = f ? { t: 'n', f: f, v: v } : (typeof v === 'number' ? { t: 'n', v: v } : { t: 's', v: String(v) }); if (fmt && ws[ref]) ws[ref].z = fmt; };
      var pliF = function (row) { return kind === 'MB' ? '(B' + row + '-C' + row + ')/B' + row : 'D' + row + '/B' + row; };
      var pliV = function (x) { return kind === 'MB' ? (x.s ? (x.s - x.c) / x.s : 0) : (x.s ? x.op / x.s : 0); };
      // encabezados
      setC('A1', 'RANGO INTERCUARTIL — MÉTODO DEL DECRETO 1625/2016, Art. 1.2.2.2.4.5');

      setC('A4', 'Comparable'); setC('B4', 'Ventas'); setC('C4', 'Costo'); setC('D4', 'Utilidad Op.'); setC('E4', 'PLI (fórmula)');
      var r0 = 5;
      comps.forEach(function (x, i) {
        var r = r0 + i;
        setC('A' + r, x.name); setC('B' + r, x.s); setC('C' + r, x.c); setC('D' + r, x.op);
        setC('E' + r, pliV(x), pliF(r), '0.0000');
      });
      var rEnd = r0 + n - 1, rng = 'E' + r0 + ':E' + rEnd;
      var R = rEnd + 2;
      // n y posiciones del decreto
      setC('A' + R, 'n (comparables)'); setC('B' + R, n, 'COUNT(' + rng + ')');

      // interpolación posicional con SMALL: v = SMALL(rng,INT(p)) + (p-INT(p))*(SMALL(rng,MIN(INT(p)+1,n))-SMALL(rng,INT(p)))
      var interp = function (pCell) { return 'SMALL(' + rng + ',INT(' + pCell + '))+(' + pCell + '-INT(' + pCell + '))*(SMALL(' + rng + ',MIN(INT(' + pCell + ')+1,B' + R + '))-SMALL(' + rng + ',INT(' + pCell + ')))'; };
      // valores JS del decreto (para caché v)
      var sorted = comps.map(pliV).sort(function (a, b) { return a - b; });
      var pos = function (p1) { var idx = p1 - 1, lo = Math.floor(idx), hi = Math.min(lo + 1, n - 1); return sorted[lo] + (sorted[hi] - sorted[lo]) * (idx - lo); };
      var med1 = (n + 1) / 2, p251 = (med1 + 1) / 2, p751 = med1 - 1 + p251;
      var vMed = pos(med1), vP25 = pos(p251), vP75 = pos(p751);
      setC('A' + (R + 5), 'P25 (decreto)'); setC('B' + (R + 5), vP25, interp('B' + (R + 2)), '0.0000');

      setC('C' + (R + 5), vP25, 'PERCENTILE.INC(' + rng + ',0.25)', '0.0000'); setC('D' + (R + 5), '← verificación PERCENTILE.INC');
      setC('C' + (R + 6), vMed, 'PERCENTILE.INC(' + rng + ',0.5)', '0.0000');
      setC('C' + (R + 7), vP75, 'PERCENTILE.INC(' + rng + ',0.75)', '0.0000');
      // parte examinada
      var R2 = R + 9;
      setC('A' + R2, 'PARTE EXAMINADA'); setC('A' + (R2 + 1), 'Ventas'); setC('B' + (R2 + 1), t_s);
      setC('A' + (R2 + 2), 'Costo'); setC('B' + (R2 + 2), t_c);
      setC('A' + (R2 + 3), 'Utilidad Operacional'); setC('B' + (R2 + 3), t_op);
      var tPLI = kind === 'MB' ? (t_s ? (t_s - t_c) / t_s : 0) : (t_s ? t_op / t_s : 0);
      var tF = kind === 'MB' ? '(B' + (R2 + 1) + '-B' + (R2 + 2) + ')/B' + (R2 + 1) : 'B' + (R2 + 3) + '/B' + (R2 + 1);
      setC('A' + (R2 + 4), 'PLI parte examinada'); setC('B' + (R2 + 4), tPLI, tF, '0.0000');
      // posición y ajuste
      var cP25 = 'B' + (R + 5), cMed = 'B' + (R + 6), cP75 = 'B' + (R + 7), cT = 'B' + (R2 + 4), cS = 'B' + (R2 + 1);
      setC('A' + (R2 + 5), '¿Dentro del rango?'); ws['B' + (R2 + 5)] = { t: 's', v: (tPLI >= vP25 && tPLI <= vP75) ? 'SÍ' : 'NO', f: 'IF(AND(' + cT + '>=' + cP25 + ',' + cT + '<=' + cP75 + '),"SÍ","NO")' };
      var ajuste = (tPLI < vP25) ? (vMed - tPLI) * t_s : 0;
      setC('A' + (R2 + 6), 'AJUSTE a la mediana (si PLI<P25)'); setC('B' + (R2 + 6), ajuste, 'IF(' + cT + '<' + cP25 + ',(' + cMed + '-' + cT + ')*' + cS + ',0)', '#,##0');
      setC('A' + (R2 + 7), 'Utilidad Op. ajustada'); setC('B' + (R2 + 7), t_op + ajuste, 'B' + (R2 + 3) + '+B' + (R2 + 6), '#,##0');
      setC('A' + (R2 + 8), 'PLI post-ajuste (→ mediana)'); setC('B' + (R2 + 8), t_s ? ((t_op + ajuste) / t_s) : 0, 'B' + (R2 + 7) + '/' + cS, '0.0000');
      ws['!ref'] = 'A1:F' + (R2 + 9);
      ws['!cols'] = [{ wch: 34 }, { wch: 16 }, { wch: 16 }, { wch: 16 }, { wch: 12 }, { wch: 28 }];
      XLSX.utils.book_append_sheet(A, ws, 'Rango Intercuartil');
      return { wb: A, meta: { n: n, kind: kind, vP25: vP25, vMed: vMed, vP75: vP75, tPLI: tPLI, ajuste: ajuste } };

  <script>/* ============================================================================
MÓDULO OCR REAL + ENRUTADOR  ·  namespace ORQ_OCR
No toca el motor PT. Se apoya en pdf.js (ya presente) y tesseract.js (CDN).
Estrategia para "mezcla de los tres":
1) PDF nativo (con capa de texto)  → extrae texto directo (rápido, sin OCR)
2) PDF escaneado (sin texto)       → rasteriza a canvas y aplica OCR real
3) Origen remoto                   → no accesible desde el navegador;
se guía la descarga manual
============================================================================ */
    window.ORQ_OCR = {
      worker: null,
      UMBRAL_TEXTO: 100,     // < 100 chars útiles ⇒ se considera escaneado
      _log: (msg) => { const el = document.getElementById('ocrStatus'); if (el) el.textContent = msg; },

      /* --- Carga perezosa de tesseract.js (solo si se necesita OCR) --- */
      async _ensureTesseract() {
        if (window.Tesseract) return true;
        this._log('Cargando motor OCR (primera vez)…');
        await new Promise((res, rej) => {
          const s = document.createElement('script');
          s.src = 'https://cdnjs.cloudflare.com/ajax/libs/tesseract.js/5.1.1/tesseract.min.js';
          s.onload = res; s.onerror = () => rej(new Error('No se pudo cargar tesseract.js'));
          document.head.appendChild(s);
        });
        return !!window.Tesseract;
      },

      async _ensurePdfjs() {
        if (!window.pdfjsLib) {
          await new Promise((res, rej) => {
            const s = document.createElement('script');
            s.src = 'vendor/pdfjs/pdf.min.js';
            s.onload = res; s.onerror = () => rej(new Error('No se pudo cargar pdf.js'));
            document.head.appendChild(s);
          });
        }
        if (window.pdfjsLib && window.pdfjsLib.GlobalWorkerOptions && !window.pdfjsLib.GlobalWorkerOptions.workerSrc) {
          window.pdfjsLib.GlobalWorkerOptions.workerSrc = 'vendor/pdfjs/pdf.worker.min.js';
        }
        return !!window.pdfjsLib;
      },

      /* --- PASO 1: clasificar el PDF (nativo vs escaneado) --- */
      async clasificar(file) {
        await this._ensurePdfjs();
        const buf = await file.arrayBuffer();
        const pdf = await window.pdfjsLib.getDocument({ data: buf }).promise;
        let texto = '';
        const maxPag = Math.min(pdf.numPages, 3);          // muestrea 3 págs para decidir
        for (let i = 1; i <= maxPag; i++) {
          const pg = await pdf.getPage(i);
          const c = await pg.getTextContent();
          texto += c.items.map(it => it.str).join(' ');
        }
        const utiles = texto.replace(/\s+/g, '').length;
        return {
          tipo: utiles >= this.UMBRAL_TEXTO ? 'nativo' : 'escaneado',
          charsMuestra: utiles, numPages: pdf.numPages, _pdf: pdf, _buf: buf
        };
      },

      /* --- PASO 2a: extracción directa (PDF nativo) --- */
      async extraerTextoNativo(pdf) {
        let txt = '';
        for (let i = 1; i <= pdf.numPages; i++) {
          const pg = await pdf.getPage(i);
          const c = await pg.getTextContent();
          txt += c.items.map(it => it.str).join(' ') + '\n';
        }
        return txt;
      },

      /* --- PASO 2b: OCR real (PDF escaneado) --- */
      async ocrEscaneado(pdf, onPag) {
        await this._ensureTesseract();
        let txt = '';
        for (let i = 1; i <= pdf.numPages; i++) {
          if (onPag) onPag(i, pdf.numPages);
          const pg = await pdf.getPage(i);
          const vp = pg.getViewport({ scale: 2.0 });        // 2x = mejor precisión OCR
          const canvas = document.createElement('canvas');
          canvas.width = vp.width; canvas.height = vp.height;
          await pg.render({ canvasContext: canvas.getContext('2d'), viewport: vp }).promise;
          const { data } = await window.Tesseract.recognize(canvas, 'spa+eng');
          txt += data.text + '\n';
          canvas.width = canvas.height = 0;                 // liberar memoria
        }
        return txt;
      },

      /* --- ORQUESTA: recibe un File local y decide la ruta --- */
      async procesar(file) {
        try {
          if (!file || file.type !== 'application/pdf') {
            return { ok: false, motivo: 'No es un PDF' };
          }
          this._log('Analizando PDF…');
          const info = await this.clasificar(file);

          if (info.tipo === 'nativo') {
            this._log(`PDF nativo (${info.numPages} pág.) · extrayendo texto…`);
            const texto = await this.extraerTextoNativo(info._pdf);
            return {
              ok: true, ruta: 'texto-directo', tipo: 'nativo',
              paginas: info.numPages, texto, ocr: false
            };
          } else {
            this._log(`PDF escaneado detectado · aplicando OCR real…`);
            const texto = await this.ocrEscaneado(info._pdf,
              (p, t) => this._log(`OCR página ${p}/${t}…`));
            return {
              ok: true, ruta: 'ocr', tipo: 'escaneado',
              paginas: info.numPages, texto, ocr: true
            };
          }
        } catch (e) {
          return {
            ok: false, motivo: e.message,
            sugerencia: 'Descargue el PDF y péguelo como texto, o intente de nuevo.'
          };
        }
      },

      /* --- Origen remoto: el navegador no puede ir a buscarlo (CORS/seguridad) --- */
      remoto(url) {
        return {
          ok: false, ruta: 'remoto-bloqueado',
          motivo: 'El navegador no puede descargar archivos de repositorios remotos por seguridad (CORS).',
          accion: 'Descargue el archivo desde ' + (url || 'el repositorio') +
            ' y adjúntelo localmente, o configure un conector Drive/SharePoint.'
        };
      }
    };

      function _numEEFF(v) {
        if (typeof v === 'number') return isFinite(v) ? v : null;
        var s = T(v); if (!s) return null;
        // «num()» ya distingue miles-con-punto de decimales-con-coma (y viceversa);
        // parseFloat directo truncaba "2.761.202.249" en 2.761 al toparse con el
        // segundo punto. Solo hace falta quitar el símbolo de moneda antes.
        s = s.replace(/\$/g, '');
        var n = (typeof num === 'function') ? num(s) : parseFloat(s.replace(/[^0-9.,\-]/g, ''));
        if (n === null || !isFinite(n) || n === 0) return null;
        return n;
      }

      /* Busca en filas de Excel: la etiqueta y el PRIMER importe a su derecha
         (columna del año corriente en un EEFF comparativo). */
      function buscarEnFilas(filas, rx) {
        for (var i = 0; i < filas.length; i++) {
          var f = filas[i] || [];
          var etiqueta = -1;
          for (var c = 0; c < f.length; c++) {
            var celda = String(f[c]).trim();
            if (rx.test(celda)) { etiqueta = c; break; }
          }
          if (etiqueta < 0) continue;
          for (var d = etiqueta + 1; d < f.length; d++) {
            var n = _numEEFF(f[d]);
            if (n !== null) return { valor: n, fila: i + 1 };
          }
        }
        return null;
      }

      /* Busca en texto plano (PDF transcrito o cuadro de copia fiel):
         toma el primer número de la línea que contiene la etiqueta. */
      function buscarEnTexto(texto, rx) {
        var lineas = String(texto).split(/\r?\n/);
        for (var i = 0; i < lineas.length; i++) {
          var ln = lineas[i];
          if (!rx.test(ln)) continue;
          var resto = ln.replace(rx, ' ');
          var m = resto.match(/\(?-?\$?\s?\d[\d.,]{2,}\)?/g);
          if (m && m.length) {
            var n = _numEEFF(m[0]);
            if (n !== null) return { valor: n, fila: i + 1 };
          }
        }
        return null;
      }

      /* Texto nativo de un PDF (sin rasterizar): necesario para leer los EEFF */
      window.ptPdfTexto = async function (file) {
        if (!window.pdfjsLib) { try { await loadScript('vendor/pdfjs/pdf.min.js'); } catch (e) { } }
        if (!window.pdfjsLib) throw new Error('Motor PDF no disponible');
        if (pdfjsLib.GlobalWorkerOptions && !pdfjsLib.GlobalWorkerOptions.workerSrc)
          pdfjsLib.GlobalWorkerOptions.workerSrc = 'vendor/pdfjs/pdf.worker.min.js';
        var pdf = await pdfjsLib.getDocument({ data: await file.arrayBuffer() }).promise;
        var t = '';
        for (var i = 1; i <= pdf.numPages; i++) {
          var pg = await pdf.getPage(i);
          var tc = await pg.getTextContent();
          // Unir todos los ítems de la página con un solo espacio (versión anterior)
          // convierte cada PÁGINA en una única "línea" para buscarEnTexto(), que
          // busca el primer número DESPUÉS de la etiqueta en la misma línea: con una
          // página entera como línea, ese "primer número" termina siendo cualquier
          // cifra que aparezca antes en el flujo del contenido (p. ej. el año del
          // encabezado "A DIC 2025"), no la cifra de la fila correcta. Se reconstruye
          // por fila visual (misma función que usa readPDF) para que cada línea sea
          // realmente una fila del estado financiero.
          t += (typeof pdfExtraerTextoPosicional === 'function' ? pdfExtraerTextoPosicional(tc.items) : tc.items.map(function (x) { return x.str; }).join(' ')) + '\n';
        }
        return t;
      };

      /* Los EEFF en PDF se siguen incorporando como imagen (conservan firmas y

      var _adjCargarPrev = window.adjCargar;
      window.adjCargar = function (files, tipo, onDone) {
        try {
          if (tipo === 'eeff' && files && files.length) {

      function puntuar(cand, perfil) {
        var base = ((cand.desc || '') + ' ' + (cand.sic || '') + ' ' + (cand.name || '')).toLowerCase();
        var s = 0, hits = [];
        var vistos = {};
        perfil.tokens.forEach(function (t) {
          if (vistos[t]) return; vistos[t] = 1;
          if (base.indexOf(t) >= 0) { s += 1; hits.push(t); }
        });
        perfil.sics.forEach(function (c) {
          if ((cand.sic || '').indexOf(c) >= 0) { s += 4; hits.push('SIC ' + c); }
          else if ((cand.sic || '').indexOf(c.slice(0, 3)) >= 0) { s += 2; hits.push('SIC ' + c.slice(0, 3) + 'x'); }
        });
        if (cand.s === null || cand.op === null) s -= 3;           // sin datos financieros
        if (cand.controlada) s -= 6;                                // no independiente
        if (/operating/i.test(cand.estado || '')) s += 1;
        return { score: s, hits: hits.slice(0, 8) };
      }

         1. LECTOR DE .DOCX SIN DEPENDENCIAS
         Un .docx es un ZIP; se localiza word/document.xml en el directorio
         central, se descomprime con DecompressionStream y se limpia el XML.
         ─────────────────────────────────────────────────────────────────────── */
      async function ptLeerDocx(file) {
        var buf = new Uint8Array(await file.arrayBuffer());
        var dv = new DataView(buf.buffer);

        /* End of Central Directory */
        var eocd = -1;
        for (var i = buf.length - 22; i >= 0 && i > buf.length - 66000; i--) {
          if (dv.getUint32(i, true) === 0x06054b50) { eocd = i; break; }
        }
        if (eocd < 0) throw new Error('El archivo no es un .docx válido (no se encontró el índice del ZIP).');

        var nEntradas = dv.getUint16(eocd + 10, true);
        var offCD = dv.getUint32(eocd + 16, true);
        var p = offCD, objetivo = null;
        var dec = new TextDecoder('utf-8');

        for (var e = 0; e < nEntradas; e++) {
          if (dv.getUint32(p, true) !== 0x02014b50) break;
          var metodo = dv.getUint16(p + 10, true);
          var csize = dv.getUint32(p + 20, true);
          var nameLen = dv.getUint16(p + 28, true);
          var extraLen = dv.getUint16(p + 30, true);
          var commLen = dv.getUint16(p + 32, true);
          var localOff = dv.getUint32(p + 42, true);
          var nombre = dec.decode(buf.subarray(p + 46, p + 46 + nameLen));
          if (nombre === 'word/document.xml') { objetivo = { metodo: metodo, csize: csize, localOff: localOff }; }
          p += 46 + nameLen + extraLen + commLen;
        }
        if (!objetivo) throw new Error('El .docx no contiene word/document.xml.');

        /* cabecera local: los tamaños de nombre/extra pueden diferir del índice */
        var lo = objetivo.localOff;
        if (dv.getUint32(lo, true) !== 0x04034b50) throw new Error('Cabecera local del ZIP inválida.');
        var lNameLen = dv.getUint16(lo + 26, true), lExtraLen = dv.getUint16(lo + 28, true);
        var ini = lo + 30 + lNameLen + lExtraLen;
        var datos = buf.subarray(ini, ini + objetivo.csize);

        var xml;
        if (objetivo.metodo === 0) {
          xml = dec.decode(datos);
        } else if (objetivo.metodo === 8) {
          if (typeof DecompressionStream === 'undefined') throw new Error('El navegador no permite descomprimir el .docx: guarde el informe como PDF y cárguelo así.');
          var ds = new DecompressionStream('deflate-raw');
          var stream = new Blob([datos]).stream().pipeThrough(ds);
          xml = await new Response(stream).text();
        } else {
          throw new Error('Método de compresión no soportado en el .docx.');
        }

        /* XML → texto con saltos de párrafo */
        return xml
          .replace(/<w:tab[^>]*\/>/g, '\t')
          .replace(/<\/w:p>/g, '\n')
          .replace(/<w:br[^>]*\/>/g, '\n')
          .replace(/<[^>]+>/g, '')
          .replace(/&lt;/g, '<').replace(/&gt;/g, '>').replace(/&amp;/g, '&')
          .replace(/&quot;/g, '"').replace(/&apos;/g, "'")
          .replace(/\n{3,}/g, '\n\n')
          .trim();
      }

      function esTitulo(linea) {
        var l = linea.trim();
        if (!l || l.length > 130) return false;
        if (/[.;]$/.test(l) && l.length > 60) return false;
        return true;
      }

      function ptSegmentar(texto) {
        var lineas = String(texto).split(/\r?\n/).map(function (l) { return l.trim(); }).filter(function (l) { return l; });
        var bloques = [], actual = null;
        lineas.forEach(function (l) {
          var esT = esTitulo(l);
          if (esT && (l.length < 90)) {
            if (actual && actual.cuerpo.length) bloques.push(actual);
            actual = { titulo: l, cuerpo: [] };
          } else if (actual) {
            actual.cuerpo.push(l);
          } else {
            actual = { titulo: '', cuerpo: [l] };
          }
        });
        if (actual && actual.cuerpo.length) bloques.push(actual);
        return bloques;
      }

      function esTOC(b) {
        /* portada / tabla de contenido / anexos: listan TODOS los títulos del informe
           anterior o describen decenas de comparables, por lo que calzan con varias
           expresiones del MAPA a la vez y contaminarían cualquier apartado con el
           mismo texto. Se detecta por los líderes de puntos ("....... 8") típicos de
           un índice, o por tratarse de un anexo, y se excluye de la clasificación. */
        var texto = b.titulo + ' ' + b.cuerpo.join(' ');
        if ((texto.match(/\.{5,}/g) || []).length >= 3) return true;
        return /anexo\b/i.test(texto.slice(0, 40).replace(/^[\s\d]+/, ''));
      }

      function ptDistribuir(texto, fuente) {
        var bloques = ptSegmentar(texto);
        var ec = {}, info = {};

        bloques.forEach(function (b) {
          if (esTOC(b)) return;
          var t = b.titulo + ' ' + b.cuerpo.slice(0, 2).join(' ');
          MAPA.forEach(function (par) {
            if (par[1].test(t)) {
              ec[par[0]] = (ec[par[0]] ? ec[par[0]] + '\n' : '') + (b.titulo ? b.titulo + '\n' : '') + b.cuerpo.join('\n');
            }
          });
          MAPA_INFOGEN.forEach(function (par) {
            if (par[1].test(t)) {
              info[par[0]] = (info[par[0]] ? info[par[0]] + ' ' : '') + b.cuerpo.join(' ');
            }
          });
        });

        /* barrido por párrafo para los apartados que ningún título cubrió */
        MAPA.forEach(function (par) {
          if (ec[par[0]]) return;
          var hits = bloques.filter(function (b) {
            return !esTOC(b) && par[1].test(b.titulo + ' ' + b.cuerpo.join(' ')) && b.cuerpo.join(' ').length > 60;
          }).slice(0, 3);
          if (hits.length) {
            ec[par[0]] = hits.map(function (b) { return (b.titulo ? b.titulo + '\n' : '') + b.cuerpo.join('\n'); }).join('\n\n');
          }
        });

        MAPA_INFOGEN.forEach(function (par) {
          if (info[par[0]]) return;
          var hit = bloques.filter(function (b) { return !esTOC(b); }).map(function (b) { return b.cuerpo.join(' '); })
            .filter(function (p) { return par[1].test(p) && p.length > 80; })[0];
          if (hit) info[par[0]] = hit;
        });

        /* volcado a los campos del formulario que estén vacíos */
        var puestos = [];
        Object.keys(ec).forEach(function (id) {
          var el = $(id);
          if (el && !String(el.value || '').trim()) {
            el.value = ec[id].slice(0, 4000);
            puestos.push(id);
          }
        });

        Object.keys(info).forEach(function (k) { INFOGEN[k] = info[k].slice(0, 2500); });
        INFOGEN._fuente = fuente;

        return { ec: ec, info: info, puestos: puestos, bloques: bloques.length };
      }

      function ptComparablesDelTexto(texto) {
        var out = [], vistos = {};
        var rx = /^([A-ZÁÉÍÓÚÑ0-9][A-Za-zÁÉÍÓÚÑáéíóúñ0-9&.,'\- ]{4,70}?(?:\s(?:S\.?A\.?S?\.?|Inc\.?|Corp(?:oration)?\.?|Ltd\.?|Limited|PLC|N\.?V\.?|A\/S|AB|GmbH|Co\.?|Holdings?|Group|Technologies|Interactive|Entertainment|Studios?|Games?|Software)))\b/;
        String(texto).split(/\r?\n/).forEach(function (l) {
          var m = l.trim().match(rx);
          if (!m) return;
          var nombre = m[1].trim();
          if (nombre.length < 6) return;
          var k = nombre.toUpperCase().replace(/[^A-Z0-9]/g, '');
          if (vistos[k]) return; vistos[k] = 1;
          var idm = l.match(/\bIQ\d{4,}\b/);
          out.push({ name: nombre, id: idm ? idm[0] : '', sic: (l.match(/\b\d{4}\b/) || [''])[0], desc: '' });
        });
        return out;

      /* Identidad: primero heurística local, luego IA si el entorno la expone */
      window.ptIdentidadDesdeAnterior = async function (texto) {
        var vacio = function (id) { var e = $(id); return e && !String(e.value || '').trim(); };
        var poner = function (id, v) { var e = $(id); if (e && v && !String(e.value || '').trim()) { e.value = String(v).trim(); return true; } return false; };
        var puestos = [];

        /* — extracción local — */
        var mNit = texto.match(/nit[^0-9]{0,12}(\d[\d.\-]{6,15})/i);
        if (mNit && poner('nit', mNit[1].replace(/[^0-9\-]/g, ''))) puestos.push('NIT');
        var mEnt = texto.match(/\b([A-ZÁÉÍÓÚÑ][A-ZÁÉÍÓÚÑ0-9&.\- ]{5,60}(?:S\.?A\.?S\.?|LTDA\.?|S\.?A\.?))\b/);
        if (mEnt && poner('ent', mEnt[1])) puestos.push('razón social');
        var mVinc = texto.match(/vinculad[oa][^.]{0,60}?\b([A-ZÁÉÍÓÚÑ][A-Za-zÁÉÍÓÚÑ0-9&.\- ]{4,50}(?:INC\.?|CORP\.?|LLC|LTD\.?|GMBH))\b/);
        if (mVinc && poner('vinc', mVinc[1])) puestos.push('vinculado');
        if (/margen operacional|\bMO\b/i.test(texto) && vacio('pli')) { poner('pli', 'MO'); puestos.push('indicador'); }

        if (puestos.length) { try { calc(); } catch (e) { } docMsg('Del informe anterior se tomó: ' + puestos.join(', ') + '.'); }

        /* — complemento con IA, si está disponible — */
        if (typeof claudeAsk !== 'function') return puestos;
        try {
          var r = await claudeAsk('Del siguiente extracto de un Informe Local de Precios de Transferencia devuelve SOLO JSON ' +
            '{"entidad":"","nit":"","vinculado":"","pais":"","operacion":"","pli":"MO|MB|Berry"} sin texto adicional.\n\n' +
            String(texto).slice(0, 12000), { useWeb: false, maxTokens: 800 });
          var j = JSON.parse(String(r.texto).replace(/```json|```/g, '').trim());
          var extra = [];
          if (poner('ent', j.entidad)) extra.push('razón social');
          if (poner('nit', j.nit)) extra.push('NIT');
          if (poner('vinc', j.vinculado)) extra.push('vinculado');
          if (poner('pais', j.pais)) extra.push('país');
          if (poner('oper', j.operacion)) extra.push('operación');
          if (j.pli && vacio('pli')) { poner('pli', j.pli); extra.push('indicador'); }
          if (extra.length) { try { calc(); } catch (e) { } docMsg('Complemento por IA desde el informe anterior: ' + extra.join(', ') + '.'); }
          return puestos.concat(extra);
        } catch (e) { return puestos; }
      };
